use std::process::ExitCode;

fn main() -> ExitCode {
    let Some(path) = std::env::args().nth(1) else {
        eprintln!("usage: qa-scen <file.pdf>");
        return ExitCode::from(2);
    };
    let pdf = match std::fs::read(&path) {
        Ok(b) => b,
        Err(e) => {
            eprintln!("read error: {e}");
            return ExitCode::from(2);
        }
    };
    println!("file: {path} ({} bytes)", pdf.len());
    match nacre_validate::validate(&pdf) {
        Ok(sigs) => {
            println!("signatures: {}", sigs.len());
            for s in &sigs {
                println!("- field {:?}", s.field);
                println!("  indication: {} / {:?}", s.report.indication().as_str(),
                    s.report.sub_indication().map(|x| x.as_str()));
                println!("  digest_verified={} signature_verified={}", s.digest_verified, s.signature_verified);
                println!("  defect: {:?}", s.defect);
                println!("  later_revisions: {}", s.later_revisions.len());
                for r in &s.later_revisions {
                    println!("    end={} changed={:?}", r.end, r.changed);
                    println!("      verdict={:?}", r.verdict);
                }
            }
            ExitCode::SUCCESS
        }
        Err(e) => {
            println!("error: {e} (kind {:?})", e.kind());
            println!("debug: {e:?}");
            ExitCode::from(1)
        }
    }
}

// Appends one invisible signature made with nacre_pades::sign to argv[1] and writes argv[2].
use std::process::ExitCode;
use nacre_core::TestSigner;
use nacre_pades::{DateTime, sign};

fn main() -> ExitCode {
    let args: Vec<String> = std::env::args().collect();
    if args.len() != 3 {
        eprintln!("usage: qa-sign <in.pdf> <out.pdf>");
        return ExitCode::from(2);
    }
    let pdf = match std::fs::read(&args[1]) {
        Ok(b) => b,
        Err(e) => { eprintln!("read error: {e}"); return ExitCode::from(2); }
    };
    let signer = match TestSigner::from_secret(&[0x22; 32]) {
        Ok(s) => s,
        Err(e) => { eprintln!("signer error: {e}"); return ExitCode::from(1); }
    };
    let time = match DateTime::new(2026, 10, 23, 9, 0, 0) {
        Ok(t) => t,
        Err(e) => { eprintln!("time error: {e}"); return ExitCode::from(1); }
    };
    match sign(&pdf, &signer, time) {
        Ok(out) => {
            if let Err(e) = std::fs::write(&args[2], &out) { eprintln!("write error: {e}"); return ExitCode::from(2); }
            println!("wrote {} ({} bytes)", args[2], out.len());
            ExitCode::SUCCESS
        }
        Err(e) => { println!("sign error: {e}"); ExitCode::from(1) }
    }
}

import sys
src, dst, old, new = sys.argv[1], sys.argv[2], sys.argv[3].encode(), sys.argv[4].encode()
assert len(old) == len(new)
b = open(src, 'rb').read()
i = b.rindex(old)
open(dst, 'wb').write(b[:i] + new + b[i+len(old):])
print(f"{dst}: replaced last {old!r} at offset {i} with {new!r}")
