// `forge` is the test bench's fictitious PKI builder, used here only to build input files.

type R = Result<Vec<SignatureValidation>, Error>;

fn iso(t: SystemTime) -> String {
    let s = t.duration_since(UNIX_EPOCH).unwrap().as_secs();
    format!("{}", DateTime::from_unix_duration(Duration::from_secs(s)).unwrap())
}
fn show(label: &str, r: &R) {
    match r {
        Err(e) => println!("    [{label}] Err({e:?}) kind={:?} display=\"{e}\"", e.kind()),
        Ok(v) => {
            if v.is_empty() { println!("    [{label}] no signature"); }
            for s in v {
                println!("    [{label}] {} indication={:?} sub={:?} defect={:?} chain={:?} bst={}",
                    s.field, s.report.indication(), s.report.sub_indication(), s.defect,
                    s.chain.as_ref().map(|c| c.certificates.len()),
                    s.best_signature_time.map_or("None".into(), iso));
            }
        }
    }
}
fn d(c: &Certificate) -> Vec<u8> { c.to_der().unwrap() }

struct Pki { root: Certificate, rk: SigningKey, int: Certificate, ik: SigningKey, signer: Certificate, sk: SigningKey,
             tsaca: Certificate, tck: SigningKey, tsa: Certificate, tk: SigningKey }
impl Pki {
    fn new() -> Self {
        let rk = key(&[0x91; 32]); let ik = key(&[0x92; 32]); let sk = key(&[0x93; 32]);
        let tck = key(&[0x94; 32]); let tk = key(&[0x95; 32]);
        let root = issue(Role::Ca, "CN=QA42 root,O=QA,C=FR", 1, &rk, None, &rk);
        let int = issue(Role::Ca, "CN=QA42 signer CA,O=QA,C=FR", 2, &ik, Some(&root), &rk);
        let signer = issue(Role::Signer, "CN=QA42 signer,O=QA,C=FR", 3, &sk, Some(&int), &ik);
        let tsaca = issue(Role::Ca, "CN=QA42 TSA CA,O=QA,C=FR", 4, &tck, Some(&root), &rk);
        let tsa = issue(Role::Tsa, "CN=QA42 TSA,O=QA,C=FR", 5, &tk, Some(&tsaca), &tck);
        Self { root, rk, int, ik, signer, sk, tsaca, tck, tsa, tk }
    }
    fn chain_signer(&self) -> ChainSigner { ChainSigner { key: self.sk.clone(), chain: vec![d(&self.signer), d(&self.int)] } }
    fn tsa(&self) -> ForgedTsa { ForgedTsa { key: self.tk.clone(), certificates: vec![d(&self.tsa), d(&self.tsaca)],
        gen_time: DateTime::new(2026, 10, 22, 8, 0, 0).unwrap(), imprint: None } }
    fn bt(&self) -> Vec<u8> { signed(&self.chain_signer(), Level::T { tsa: &self.tsa() }) }
    fn bb(&self) -> Vec<u8> { signed(&self.chain_signer(), Level::B) }
    fn int_crl(&self, at: Time, revoked: &[(&Certificate, Time)]) -> Vec<u8> { crl(&self.int, &self.ik, at, time(2026, 11, 22, 0, 0, 0), revoked) }
    fn tsaca_crl(&self, at: Time, revoked: &[(&Certificate, Time)]) -> Vec<u8> { crl(&self.tsaca, &self.tck, at, time(2026, 11, 22, 0, 0, 0), revoked) }
    fn root_crl(&self, at: Time, revoked: &[(&Certificate, Time)]) -> Vec<u8> { crl(&self.root, &self.rk, at, time(2026, 11, 22, 0, 0, 0), revoked) }
    // "sig-ts": a B-T with `crls` in its /DSS; "doc-ts": a B-B with `crls` in its /DSS, then a document timestamp.
    fn file(&self, kind: &str, crls: &[Vec<u8>]) -> Vec<u8> {
        match kind {
            "sig-ts" => with_dss(&self.bt(), &[], crls),
            _ => document_timestamp(&with_dss(&self.bb(), &[], crls), &self.tsa()),
        }
    }
    fn judge(&self, pdf: &[u8], at: SystemTime, policy: &ValidationPolicy, source: Option<&dyn RevocationSource>) -> R {
        let root = d(&self.root);
        let anchors = TrustAnchors::from_certificates([root.as_slice()]).unwrap();
        validate_with_policy(pdf, &anchors, at, source, policy)
    }
}

fn control() -> SystemTime { instant(2026, 10, 23, 0, 0, 0) }

fn stale() -> Time { time(2026, 10, 21, 23, 0, 0) }
fn no_fresh() -> ValidationPolicy { let mut p = ValidationPolicy::strict(); p.require_fresh_revocation = false; p }
fn policies(which: &str) -> Vec<(&'static str, ValidationPolicy)> {
    match which {
        "strict" => vec![("strict", ValidationPolicy::strict())],
        _ => vec![("dss", ValidationPolicy::dss()), ("strict minus require_fresh_revocation", no_fresh()),
                  ("dss plus require_fresh_revocation", { let mut p = ValidationPolicy::dss(); p.require_fresh_revocation = true; p })],
    }
}

// Answers a CRL by issuer: (signer CA der, its CRL, TSA CA der, its CRL, root CRL).
struct Answer(Vec<u8>, Vec<u8>, Vec<u8>, Vec<u8>, Vec<u8>);
impl RevocationSource for Answer {
    fn revocation(&self, _c: &[u8], issuer: &[u8]) -> Result<RevocationData, nacre_core::Error> {
        if issuer == self.0 { Ok(RevocationData::Crl(self.1.clone())) }
        else if issuer == self.2 { Ok(RevocationData::Crl(self.3.clone())) }
        else { Ok(RevocationData::Crl(self.4.clone())) }
    }
}

        let files = [("sig-ts", with_dss_ocsps(&p.bt(), &[], &crls, &[resp.clone()])),
                     ("doc-ts", document_timestamp(&with_dss_ocsps(&p.bb(), &[], &crls, &[resp.clone()]), &p.tsa()))];
        for (kind, pdf) in &files {
            for (label, policy) in [("strict", ValidationPolicy::strict()), ("dss", ValidationPolicy::dss()), ("strict minus require_fresh_revocation", no_fresh())] {
                show(&format!("{kind} {label}"), &p.judge(pdf, control(), &policy, None));
            }
        }
    }
}

fn main() {
    match std::env::args().nth(1).as_deref() {
        Some("c1") => run_table("strict"),
        Some("c1-error") => c1_error(),
        Some("c2") => run_table("dss"),
        Some("c2-corpus") => c2_corpus(),
        Some("c4") => c4_ocsp(),
        other => { eprintln!("usage: qa-scen c1|c1-error|c2|c2-corpus|c4 (got {other:?})"); std::process::exit(2) }
    }
}
