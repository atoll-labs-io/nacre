use nacre_pdf::{Dictionary, Document, Object, ObjectId, SignatureSpec};
use std::process::exit;

fn short(o: &Object) -> String {
    let s = format!("{o:?}");
    s.chars().take(400).collect()
}

fn summary(label: &str, bytes: &[u8]) -> Option<()> {
    match Document::parse(bytes) {
        Ok(doc) => {
            println!("{label}: reparse ok, {} bytes, {} revisions", bytes.len(), doc.revisions().len());
            for (i, r) in doc.revisions().iter().enumerate() {
                println!("  revision {i}: end={} startxref={} xref={:?}", r.end, r.startxref, r.xref);
            }
            for f in doc.signature_fields() {
                println!("  field {:?} {} /V {:?}", f.name, f.id, f.value.map(|v| v.to_string()));
            }
            Some(())
        }
        Err(e) => { println!("{label}: reparse ERROR: {e}"); None }
    }
}

fn spec(args: &[String]) -> SignatureSpec {
    let mut entries = Dictionary::new();
    if let Some(extra) = args.get(0) {
        for kv in extra.split(',').filter(|s| !s.is_empty()) {
            let (k, v) = kv.split_once('=').unwrap_or((kv, ""));
            entries.insert(k.as_bytes(), Object::Name(v.as_bytes().to_vec()));
        }
    }
    SignatureSpec {
        field: std::env::var("FIELD").unwrap_or("Nacre1".into()),
        sub_filter: std::env::var("SUBFILTER").unwrap_or("ETSI.CAdES.detached".into()).into_bytes(),
        contents_len: std::env::var("LEN").ok().and_then(|s| s.parse().ok()).unwrap_or(8192),
        entries,
    }
}
