use nacre_core::{
    DigestAlgorithm, Error, Indication, RevocationData, RevocationSource, SignatureAlgorithm,
    SignatureForm, SignatureLevel, SignatureProfile, Signer, SubIndication, TestSigner,
    TimestampAuthority, ValidationReport,
};
use p256::ecdsa::signature::hazmat::PrehashVerifier;
use p256::ecdsa::{Signature, VerifyingKey};
use sha2::{Digest, Sha256};
use der::Decode;

struct MyTsa;
impl TimestampAuthority for MyTsa {
    fn timestamp(&self, algorithm: DigestAlgorithm, imprint: &[u8]) -> Result<Vec<u8>, Error> {
        if imprint.len() != algorithm.output_len() {
            return Err(Error::DigestLength { expected: algorithm.output_len(), actual: imprint.len() });
        }
        Err(Error::Timestamp(Box::new(std::io::Error::other("tsa unreachable"))))
    }
}
struct MyRevocation;
impl RevocationSource for MyRevocation {
    fn revocation(&self, _certificate: &[u8], _issuer: &[u8]) -> Result<RevocationData, Error> {
        Ok(RevocationData::Crl(vec![0x30, 0x00]))
    }
}

fn main() {
    // Criterion 2: the three traits, used as trait objects, synchronous.
    let signer = TestSigner::from_secret(&[7u8; 32]).expect("signer");
    let s: &dyn Signer = &signer;
    let tsa: &dyn TimestampAuthority = &MyTsa;
    let rev: &dyn RevocationSource = &MyRevocation;
    println!("algorithm = {:?}, digest = {:?}", s.algorithm(), s.algorithm().digest());
    println!("chain length = {}, first cert {} bytes", s.certificate_chain().len(), s.certificate_chain()[0].len());
    let digest = Sha256::digest(b"hello nacre");
    match tsa.timestamp(DigestAlgorithm::Sha256, &digest) {
        Err(e) => println!("tsa error: {e}; source: {:?}", std::error::Error::source(&e).map(|s| s.to_string())),
        Ok(_) => println!("tsa ok"),
    }
    println!("revocation = {:?}", rev.revocation(&s.certificate_chain()[0], &s.certificate_chain()[0]));

    // Criterion 3: the report.
    let lvl = SignatureLevel { form: SignatureForm::PAdES, profile: SignatureProfile::BaselineB };
    for r in [
        ValidationReport::passed(Some(lvl)),
        ValidationReport::failed(SubIndication::HashFailure, None),
        ValidationReport::indeterminate(SubIndication::TryLater, Some(lvl)),
    ] {
        println!("report: {} {} sub={:?} level={:?}", r.indication().as_str(), r.indication().urn(),
            r.sub_indication().map(|x| (x.as_str(), x.urn())), r.level());
    }
    println!("sub-indications: {}", SubIndication::ALL.len());
    for x in SubIndication::ALL { print!("{} ", x.as_str()); }
    println!();
    let _ = Indication::Indeterminate;

    // Criterion 4: sign and verify with p256.
    let sig = s.sign_digest(&digest).expect("sign");
    let cert = x509_cert::Certificate::from_der(&s.certificate_chain()[0]).expect("cert");
    let spki = cert.tbs_certificate().subject_public_key_info().subject_public_key.raw_bytes();
    let vk = VerifyingKey::from_sec1_bytes(spki).expect("vk");
    let parsed = Signature::from_der(&sig).expect("der sig");
    println!("signature {} bytes DER, verify = {:?}", sig.len(), vk.verify_prehash(&digest, &parsed).is_ok());
    let mut other = digest;
    other[0] ^= 1;
    println!("verify on altered digest = {:?}", vk.verify_prehash(&other, &parsed).is_ok());
    println!("cert subject = {}", cert.tbs_certificate().subject());
    let _ = SignatureAlgorithm::RsaPkcs1v15Sha256;
}
