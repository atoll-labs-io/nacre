use nacre_core::{Signer, TestSigner};

fn show(label: &str, r: Result<Vec<u8>, nacre_core::Error>) {
    match r {
        Ok(v) => println!("{label}: Ok({} bytes)", v.len()),
        Err(e) => println!("{label}: Err(Display = \"{e}\", Debug = {e:?}, source = {:?})",
            std::error::Error::source(&e).map(|s| s.to_string())),
    }
}

fn main() {
    let signer = TestSigner::from_secret(&[7u8; 32]).expect("signer");
    show("20-byte digest (SHA-1 length)", signer.sign_digest(&[0u8; 20]));
    show("empty digest", signer.sign_digest(&[]));
    show("48-byte digest (SHA-384 length)", signer.sign_digest(&[0u8; 48]));
    for (label, secret) in [("all-zero secret", [0u8; 32]), ("all-0xFF secret (above the curve order)", [0xFFu8; 32])] {
        match TestSigner::from_secret(&secret) {
            Ok(_) => println!("{label}: Ok(signer built)"),
            Err(e) => println!("{label}: Err(Display = \"{e}\", source = {:?})",
                std::error::Error::source(&e).map(|s| s.to_string())),
        }
    }
}
