# Run from the repository copy. $1: file holding the old author's surname (never printed).
NAME=$(cat "$1")
echo "== pdfinfo Author/Title/Subject/Creator, dc:creator, pdfsig signer, over every tracked PDF"
git -C "$2" ls-files '*.pdf' | while read f; do
  a=$(pdfinfo "$f" 2>/dev/null | grep -E '^(Author|Title|Subject|Creator):' | tr -s ' ' | paste -sd ';')
  m=$(pdfinfo -meta "$f" 2>/dev/null | grep -o '<dc:creator>.*</dc:creator>' | sed 's/<[^>]*>//g' | tr -s ' ' | paste -sd ';')
  s=$(pdfsig "$f" 2>/dev/null | grep 'Common Name' | sed 's/.*Name: //' | sort -u | paste -sd ',')
  [ -n "$a$m$s" ] && echo "$f | $a | dc:creator=$m | signers=$s"
done
echo "== subjects of tracked certificates (*.der, *.pem, *.crt that parse as X.509)"
git -C "$2" ls-files '*.der' '*.pem' '*.crt' | while read f; do
  s=$(openssl x509 -noout -subject -in "$f" 2>/dev/null || openssl x509 -inform DER -noout -subject -in "$f" 2>/dev/null) && echo "$f: $s"
done | sed 's/^[^:]*: //' | sort | uniq -c
echo "== case-insensitive search of the old author's surname over every tracked file (incl. binary)"
git -C "$2" grep -I -i -c -- "$NAME" ; echo "text grep exit: $? (1 = no match)"
git -C "$2" grep -a -i -l -- "$NAME" ; echo "binary-inclusive grep exit: $? (1 = no match)"
echo "== a tracked file with the old sha256 17c74d6c…"
git -C "$2" ls-files -z | xargs -0 sha256sum | grep -c '^17c74d6c6911cb8beed4430de172256bd5bed6bd' ; echo "(count above)"
