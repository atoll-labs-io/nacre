use nacre_cms::SignedData;
use std::time::Instant;

fn run(label: &str, bytes: &[u8], content: Option<&[u8]>) {
    let start = Instant::now();
    let r = std::panic::catch_unwind(|| match SignedData::from_der(bytes) {
        Err(e) => format!("from_der Err -> \"{e}\""),
        Ok(sd) => match sd.verify(content) { Err(e) => format!("from_der Ok, verify Err -> \"{e}\""), Ok(_) => "VERIFIED".into() },
    });
    match r {
        Ok(s) => println!("{label}: {s} ({:?})", start.elapsed()),
        Err(_) => println!("{label}: PANIC"),
    }
}

fn main() {
    let args: Vec<String> = std::env::args().collect();
    let good = std::fs::read(&args[1]).expect("fixture");
    let content = std::fs::read(&args[2]).expect("content");

    run("one byte 0x30", &[0x30], Some(&content));
    run("text, not DER", b"-----BEGIN PKCS7-----\nMIIB\n-----END PKCS7-----\n", Some(&content));

    for n in 0..good.len() {

    }
    println!("every strict prefix (0..{}): {errs} Err, {oks} Ok, no panic printed above", good.len());
    let mut l = good.clone(); l[1] = 0x84; l.splice(2..2, [0x7f, 0xff, 0xff, 0xff]);

    let mut l2 = good.clone(); l2[1] = 0x89;

    let mut trailing = good.clone(); trailing.extend_from_slice(&[0, 0]);

// Criterion 4: deep nesting where the schema is open (an attribute value, a certificate, an unsigned attribute).
use der::{Any, Decode};
use nacre_cms::{Attribute, Profile, SignedAttributes, SignedData, oid, sign_detached};
use nacre_core::TestSigner;
use sha2::{Digest, Sha256};

fn deep(depth: usize) -> Vec<u8> {
    fn len(n: usize) -> Vec<u8> {
        if n < 128 { return vec![n as u8]; }
        let b = n.to_be_bytes(); let b: Vec<u8> = b.iter().copied().skip_while(|x| *x == 0).collect();
        let mut v = vec![0x80 | b.len() as u8]; v.extend(b); v
    }
    let mut inner = vec![0x05, 0x00];
    for _ in 0..depth { let mut h = vec![0x30]; h.extend(len(inner.len())); h.extend(inner); inner = h; }
    inner
}

fn report(label: &str, r: std::thread::Result<String>) {
    match r { Ok(s) => println!("{label}: {s}"), Err(_) => println!("{label}: PANIC") }
}

fn main() {
    let content = b"QA s03 deep nesting".to_vec();
    let signer = TestSigner::from_secret(&[0x42; 32]).expect("signer");
    let good = sign_detached(&Sha256::digest(&content), &signer, &Profile::CMS, None).expect("sign");
    let blob = deep(50_000);
    println!("nested blob: {} bytes, depth 50000", blob.len());

    report("Any::from_der of the blob (der itself)", std::panic::catch_unwind(|| format!("{:?}", Any::from_der(&blob).map(|a| a.value().len()).map_err(|e| e.to_string()))));

    // 1. the blob as the value of an extra signed attribute
    let r = std::panic::catch_unwind(|| {
        let mut sd = SignedData::from_der(&good).expect("decode");
        let mut attrs = sd.signer_infos[0].signed_attrs.as_ref().expect("a").attributes().expect("a");
        attrs.push(Attribute { oid: oid::SIGNING_TIME, values: vec![Any::from_der(&blob).expect("any")] });
        sd.signer_infos[0].signed_attrs = Some(SignedAttributes::from_attributes(attrs).expect("attrs"));
        let der = sd.to_der().expect("to_der");
        match SignedData::from_der(&der) { Err(e) => format!("from_der Err -> \"{e}\""), Ok(sd) => match sd.verify(Some(&content)) { Ok(_) => "VERIFIED".into(), Err(e) => format!("verify Err -> \"{e}\"") } }
    });
    report("blob as a signed attribute value", r);

    // 2. the blob in place of the certificate
    let r = std::panic::catch_unwind(|| {
        let mut sd = SignedData::from_der(&good).expect("decode");
        sd.certificates = vec![blob.clone()];
        let der = sd.to_der().expect("to_der");
        match SignedData::from_der(&der) { Err(e) => format!("from_der Err -> \"{e}\""), Ok(sd) => match sd.verify(Some(&content)) { Ok(_) => "VERIFIED".into(), Err(e) => format!("verify Err -> \"{e}\"") } }
    });
    report("blob as the certificate", r);

    // 3. the blob as an unsigned attribute: the signature still holds, by design
    let r = std::panic::catch_unwind(|| {
        let mut sd = SignedData::from_der(&good).expect("decode");
        sd.signer_infos[0].push_unsigned_attribute(Attribute { oid: oid::SIGNING_TIME, values: vec![Any::from_der(&blob).expect("any")] });
        let der = sd.to_der().expect("to_der");
        match SignedData::from_der(&der) { Err(e) => format!("from_der Err -> \"{e}\""), Ok(sd) => match sd.verify(Some(&content)) { Ok(_) => format!("from_der Ok ({} unsigned attr), verify Ok", sd.signer_infos[0].unsigned_attrs.len()), Err(e) => format!("verify Err -> \"{e}\"") } }
    });
    report("blob as an unsigned attribute value", r);
}
