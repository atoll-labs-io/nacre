// Criterion 1: digest + Signer -> detached DER SignedData with the three signed attributes.
use nacre_cms::{Profile, SignedData, oid, sign_detached};
use nacre_core::{Signer, TestSigner};
use sha2::{Digest, Sha256};

fn main() {
    let args: Vec<String> = std::env::args().collect();
    let content = std::fs::read(&args[1]).expect("read content");
    let out = &args[2];
    let signer = TestSigner::from_secret(&[0x42; 32]).expect("signer");
    let digest = Sha256::digest(&content);
    let der = sign_detached(&digest, &signer, &Profile::CMS, None).expect("sign");
    std::fs::write(out, &der).expect("write");
    std::fs::write(format!("{out}.cert.der"), &signer.certificate_chain()[0]).expect("cert");
    println!("wrote {} bytes to {out}", der.len());
    let sd = SignedData::from_der(&der).expect("decode own output");
    println!("econtent present: {}", sd.econtent().is_some());
    let si = &sd.signer_infos[0];
    for a in si.signed_attrs.as_ref().expect("signed attrs").attributes().expect("attrs") {
        let name = match a.oid {
            o if o == oid::CONTENT_TYPE => "content-type",
            o if o == oid::MESSAGE_DIGEST => "message-digest",
            o if o == oid::SIGNING_CERTIFICATE_V2 => "signing-certificate-v2",
            o if o == oid::SIGNING_TIME => "signing-time",
            _ => "other",
        };
        println!("signed attribute {} ({name}), {} value(s)", a.oid, a.values.len());
    }
    println!("verify(Some(content)) = {:?}", sd.verify(Some(&content)).map(|v| v.certificate().len()));
}
