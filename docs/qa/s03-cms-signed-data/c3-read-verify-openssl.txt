// Criterion 3: read an OpenSSL SignedData and verify it; args: <der> <content or -> [attached]
use nacre_cms::SignedData;

fn main() {
    let args: Vec<String> = std::env::args().collect();
    let der = std::fs::read(&args[1]).expect("read der");
    let content = if args[2] == "-" { None } else { Some(std::fs::read(&args[2]).expect("read content")) };
    match SignedData::from_der(&der) {
        Err(e) => { println!("from_der: Err({e:?}) -> \"{e}\""); std::process::exit(2) }
        Ok(sd) => {
            println!("from_der: Ok, {} signer(s), {} certificate(s), econtent = {:?}",
                sd.signer_infos.len(), sd.certificates.len(), sd.econtent().map(|c| c.len()));
            match sd.verify(content.as_deref()) {
                Ok(v) => println!("verify: Ok, signer certificate {} bytes", v.certificate().len()),
                Err(e) => { println!("verify: Err({e:?}) -> \"{e}\""); std::process::exit(1) }
            }
        }
    }
}
