// Criterion 5, second entry: every attack file through verify_detached_digest (the ByteRange path),
// with the right digest, a digest of the wrong length, and the digest of other content.
use nacre_cms::SignedData;
use sha2::{Digest, Sha256};

fn main() {
    let args: Vec<String> = std::env::args().collect();
    let content = std::fs::read(&args[1]).expect("content");
    let right = Sha256::digest(&content);
    let other = Sha256::digest(b"other content");
    for f in &args[2..] {
        let sd = SignedData::from_der(&std::fs::read(f).expect("file")).expect("decode");
        let show = |d: &[u8]| match sd.verify_detached_digest(d) { Ok(_) => "Ok".to_string(), Err(e) => format!("Err({e:?})") };
        println!("{f}: right digest {} | 20-byte digest {} | other digest {}", show(&right), show(&right[..20]), show(&other));
    }
}
