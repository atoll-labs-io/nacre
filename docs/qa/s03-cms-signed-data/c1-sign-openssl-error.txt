// Criterion 1, invalid input: a digest of the wrong length, and an empty digest.
use nacre_cms::{Profile, sign_detached};
use nacre_core::TestSigner;
use sha2::{Digest, Sha256};

fn main() {
    let signer = TestSigner::from_secret(&[0x42; 32]).expect("signer");
    let raw = b"not a digest, the raw content".to_vec();
    let sha1_len = [0u8; 20];
    let sha512_len = [0u8; 64];
    let ok = Sha256::digest(b"x");
    for (label, d) in [("raw content (29 bytes)", &raw[..]), ("20 bytes", &sha1_len[..]), ("64 bytes", &sha512_len[..]), ("empty", &[][..]), ("32 bytes (control)", &ok[..])] {
        match sign_detached(d, &signer, &Profile::CMS, None) {
            Ok(v) => println!("{label}: Ok({} bytes)", v.len()),
            Err(e) => println!("{label}: Err(Display = {e:?} -> \"{e}\")"),
        }
    }
}
