/// A CRL of `issuer` revoking nothing, with one CRL extension `oid` holding `value`, signed with `key`.
fn crl_ext(issuer: &Certificate, key: &SigningKey, this_update: Time, value: Vec<u8>, critical: bool) -> Vec<u8> {
    let der = crl(issuer, key, this_update, time(2026, 11, 22, 0, 0, 0), &[]);
    let mut list: CertificateList = CertificateList::from_der(&der).unwrap();
    list.tbs_cert_list.crl_extensions.get_or_insert_with(Vec::new).push(Extension {
        extn_id: ObjectIdentifier::new_unwrap("2.5.29.60"), critical, extn_value: OctetString::new(value).unwrap() });
    let tbs = list.tbs_cert_list.to_der().unwrap();
    let sig: DerSignature = key.sign_prehash(&Sha256::digest(&tbs)).unwrap();
    list.signature = BitString::from_bytes(sig.as_bytes()).unwrap();
    list.to_der().unwrap()
}

/// Offset written after the last `startxref`, and where that number starts in the file.
fn last_startxref(pdf: &[u8]) -> (usize, usize) {
    let at = pdf.windows(9).rposition(|w| w == b"startxref").unwrap() + 9;
    let start = at + pdf[at..].iter().position(|b| b.is_ascii_digit()).unwrap();
    let end = start + pdf[start..].iter().position(|b| !b.is_ascii_digit()).unwrap();
    (std::str::from_utf8(&pdf[start..end]).unwrap().parse().unwrap(), start)
}
fn set_startxref(pdf: &[u8], to: usize) -> Vec<u8> {
    let (n, start) = last_startxref(pdf);
    let (old, new) = (n.to_string(), to.to_string());
    assert_eq!(old.len(), new.len());
    let mut out = pdf.to_vec(); out[start..start + old.len()].copy_from_slice(new.as_bytes()); out
}
fn last_prev(pdf: &[u8]) -> (usize, usize) {
    let at = pdf.windows(5).rposition(|w| w == b"/Prev").unwrap() + 5;

    let end = start + pdf[start..].iter().position(|b| !b.is_ascii_digit()).unwrap();

}

// ---------- c6: eleven switches over the attack corpus and the catalogue (AC6)
const SWITCHES: [&str; 11] = ["refuse_later_revisions", "refuse_sneaky_signatures", "refuse_trailing_data", "list_unusable_signatures",
    "judge_signer_key_usage", "judge_validity_before_revocation", "judge_malformed_extensions", "require_fresh_revocation", "refuse_unsupported_sub_filters",
    "refuse_misplaced_xref", "ignore_expired_certs_on_crl"];
struct Stub { int: Vec<u8>, int_crl: Vec<u8>, root_crl: Vec<u8> }
impl RevocationSource for Stub {
    fn revocation(&self, _c: &[u8], issuer: &[u8]) -> Result<RevocationData, nacre_core::Error> {
        Ok(RevocationData::Crl(if issuer == self.int { self.int_crl.clone() } else { self.root_crl.clone() }))
    }
}
fn verdict(r: &R) -> String {
    match r {
        Err(e) => { let s = format!("{e:?}"); format!("Err({})", s.split(|c: char| !c.is_alphanumeric()).next().unwrap()) }
        Ok(v) => format!("{:?}", v.iter().map(|s| (s.field.clone(), s.report.indication(), s.report.sub_indication())).collect::<Vec<_>>()),
    }
}
fn detail(r: &R) -> String { match r { Err(e) => format!("Err({e:?})"), Ok(v) => format!("{:?}", v.iter().map(|s| (&s.field, &s.defect)).collect::<Vec<_>>()) } }
fn catalog(names: &BTreeSet<String>) -> BTreeMap<String, Vec<String>> {
    let mut cells: BTreeMap<String, Vec<String>> = BTreeMap::new();
    for doc in ["pades.md", "cms.md", "revocation.md", "tsp.md"] {
        let Ok(text) = std::fs::read_to_string(format!("{CATALOG}{doc}")) else { continue };
        for line in text.lines().filter(|l| l.starts_with('|')) {
            let parts: Vec<&str> = line.trim().trim_matches('|').split('|').map(str::trim).collect();
            let Some(case) = parts.iter().find(|c| names.contains(c.trim_matches('`'))) else { continue };
            cells.entry(case.trim_matches('`').to_string()).or_default().push(format!("{doc}: {}", parts.last().unwrap()));
        }
    }
    cells
}
fn ticks(cell: &str) -> Vec<String> { cell.split('`').skip(1).step_by(2).map(String::from).collect() }
fn c6(error: bool) {
    let chain_int = read(&format!("{VCORPUS}chain/intermediate.der"));
    let chain_root = read(&format!("{VCORPUS}chain/root.der"));
    let anchors = TrustAnchors::from_certificates([chain_root.as_slice()]).unwrap();
    let int_c = Certificate::from_der(&chain_int).unwrap(); let root_c = Certificate::from_der(&chain_root).unwrap();
    let t = time(2026, 10, 21, 12, 0, 0);
    let stub = Stub { int: chain_int.clone(), int_crl: crl(&int_c, &key(&[0x32; 32]), t, time(2026, 12, 1, 0, 0, 0), &[]), root_crl: crl(&root_c, &key(&[0x31; 32]), t, time(2026, 12, 1, 0, 0, 0), &[]) };
    let at = instant(2026, 10, 22, 0, 0, 0);
    let run = |pdf: &[u8], p: &ValidationPolicy| validate_with_policy(pdf, &anchors, at, Some(&stub), p);
    println!("Anchor chain/root.der, control date 2026-10-22T00:00Z, RevocationSource answering \"not revoked\" CRLs of the chain intermediate and root issued 2026-10-21T12:00Z (forged here with the corpus keys [0x32;32], [0x31;32] named in examples/forge_chain_attacks.rs)");
    if error {
        let mut inputs: Vec<(String, Vec<u8>)> = vec![("chain/chain-dss.pdf".into(), read(&format!("{VCORPUS}chain/chain-dss.pdf"))), ("unsigned.pdf".into(), read(&format!("{VCORPUS}unsigned.pdf"))), ("empty input".into(), vec![]), ("not a PDF".into(), b"hello".to_vec())];
        let a = read(&format!("{VCORPUS}attacks/xref-misplaced-startxref.pdf")); inputs.push(("attacks/xref-misplaced-startxref.pdf truncated to half".into(), a[..a.len() / 2].to_vec()));
        for (n, pdf) in inputs {
            let s = run(&pdf, &ValidationPolicy::strict());
            println!("{n}: strict {}  detail {}", verdict(&s), detail(&s));
            println!("  dss {}", verdict(&run(&pdf, &ValidationPolicy::dss())));
            let changed: Vec<&str> = SWITCHES.iter().copied().filter(|sw| verdict(&run(&pdf, &without(sw))) != verdict(&s)).collect();
            println!("  switches whose removal alone changes the verdict: {changed:?}");
        }
        return;
    }
    let mut files: Vec<String> = std::fs::read_dir(format!("{VCORPUS}attacks")).unwrap().map(|e| e.unwrap().file_name().into_string().unwrap()).filter(|n| n.ends_with(".pdf")).collect();
    files.sort();
    let names: BTreeSet<String> = files.iter().cloned().collect();
    let cat = catalog(&names);
    let (mut m6, mut m7, mut clean) = (0, 0, 0);
    for f in &files {
        let pdf = read(&format!("{VCORPUS}attacks/{f}"));
        let s = run(&pdf, &ValidationPolicy::strict());
        let dv = run(&pdf, &ValidationPolicy::dss());
        let changed: BTreeSet<String> = SWITCHES.iter().filter(|sw| verdict(&run(&pdf, &without(sw))) != verdict(&s)).map(|s| s.to_string()).collect();
        let opened = verdict(&dv) != verdict(&s) && matches!(&dv, Ok(v) if v.iter().any(|x| x.defect.is_none()));
        let strict_clean = matches!(&s, Ok(v) if v.iter().any(|x| x.defect.is_none()));
        let cells = cat.get(f).cloned().unwrap_or_default();
        println!("{f}\n  strict: {}  defects {}\n  dss:    {}\n  switches whose removal alone changes the verdict: {changed:?}\n  opened under dss: {opened}", verdict(&s), detail(&s), verdict(&dv));
        for c in &cells { println!("    catalog {c}"); }
        let named: BTreeSet<String> = cells.iter().flat_map(|c| ticks(c.split_once(": ").unwrap().1)).filter(|n| SWITCHES.contains(&n.as_str())).collect();
        let starts = cells.iter().any(|c| c.split_once(": ").unwrap().1.starts_with("`dss()`"));
        if cells.len() != 1 || named != changed { m6 += 1; println!("  MISMATCH isolation: catalog names {named:?}, observed {changed:?}, rows {}", cells.len()); }
        if starts != opened { m7 += 1; println!("  MISMATCH opened: catalog {starts}, observed {opened}"); }
        if strict_clean { clean += 1; println!("  a signature without defect under strict"); }
    }
    let used: BTreeSet<String> = cat.values().flatten().flat_map(|c| ticks(c.split_once(": ").unwrap().1)).filter(|n| SWITCHES.contains(&n.as_str())).collect();
    println!("SUMMARY files={} isolation mismatches={m6} opened mismatches={m7} files with a defect-free signature under strict={clean}", files.len());
    println!("switches named by at least one catalogue cell of a corpus file: {} of 11: {used:?}", used.len());

        for sw in ["refuse_misplaced_xref", "ignore_expired_certs_on_crl", "ADR-0029"] { println!("  {doc} mentions {sw}: {}", text.matches(sw).count()); }
    }
}
