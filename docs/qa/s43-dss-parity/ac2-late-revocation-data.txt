type R = Result<Vec<SignatureValidation>, Error>;

fn show(label: &str, r: &R) {
    match r {
        Err(e) => println!("  [{label}] Err({e:?}) kind={:?} display=\"{e}\"", e.kind()),
        Ok(v) => {
            println!("  [{label}] signatures: {}", v.len());
            for s in v {
                println!(
                    "    field={} indication={:?} sub={:?} defect={:?} digest={} sig={} chain={:?} bst={:?}",
                    s.field, s.report.indication(), s.report.sub_indication(), s.defect,
                    s.digest_verified, s.signature_verified,
                    s.chain.as_ref().map(|c| c.certificates.len()), s.best_signature_time
                );
            }
        }
    }
}

fn file(name: &str) -> Vec<u8> {
    std::fs::read(format!("{CORPUS}{name}")).unwrap_or_else(|e| panic!("{name}: {e}"))
}
fn d(c: &Certificate) -> Vec<u8> { c.to_der().unwrap() }

fn presets() -> [(&'static str, ValidationPolicy); 2] {
    [("strict", ValidationPolicy::strict()), ("dss", ValidationPolicy::dss())]

struct Pki { root: Certificate, rk: SigningKey, int: Certificate, ik: SigningKey, signer: Certificate, sk: SigningKey, tsa: Certificate, tk: SigningKey }
impl Pki {
    fn new() -> Self {
        let rk = key(&[0x61; 32]); let ik = key(&[0x62; 32]); let sk = key(&[0x63; 32]); let tk = key(&[0x64; 32]);
        let root = issue(Role::Ca, "CN=QA43 root,O=QA,C=FR", 1, &rk, None, &rk);
        let int = issue(Role::Ca, "CN=QA43 intermediate,O=QA,C=FR", 2, &ik, Some(&root), &rk);
        let signer = issue(Role::Signer, "CN=QA43 signer,O=QA,C=FR", 3, &sk, Some(&int), &ik);
        let tsa = issue(Role::Tsa, "CN=QA43 TSA,O=QA,C=FR", 4, &tk, Some(&int), &ik);
        Pki { root, rk, int, ik, signer, sk, tsa, tk }
    }
    fn anchors(&self) -> TrustAnchors { let r = d(&self.root); TrustAnchors::from_certificates([r.as_slice()]).unwrap() }
    fn leaf(&self, serial: u64, nb: Time, na: Time) -> Certificate {
        issue_valid(Role::Signer, &format!("CN=QA43 signer {serial},O=QA,C=FR"), serial, &self.sk, Some(&self.int), &self.ik, nb, na)
    }
    fn chain(&self, leaf: &Certificate) -> ChainSigner { ChainSigner { key: self.sk.clone(), chain: vec![d(leaf), d(&self.int)] } }
    fn tsa_auth(&self, cert: &Certificate) -> ForgedTsa {
        ForgedTsa { key: self.tk.clone(), certificates: vec![d(cert)], gen_time: DateTime::new(2026, 10, 22, 8, 0, 0).unwrap(), imprint: None }
    }
    fn crl_int(&self, at: Time) -> Vec<u8> { crl(&self.int, &self.ik, at, time(2026, 12, 1, 0, 0, 0), &[]) }
    fn crl_int_rev(&self, at: Time, c: &Certificate, when: Time) -> Vec<u8> { crl(&self.int, &self.ik, at, time(2026, 12, 1, 0, 0, 0), &[(c, when)]) }
    fn crl_root(&self, at: Time) -> Vec<u8> { crl(&self.root, &self.rk, at, time(2026, 12, 1, 0, 0, 0), &[]) }
    fn fresh(&self) -> Vec<Vec<u8>> { let t = time(2026, 10, 23, 12, 0, 0); vec![self.crl_int(t), self.crl_root(t)] }
    fn judge(&self, label: &str, pdf: &[u8], p: &ValidationPolicy) -> R {
        let r = validate_with_policy(pdf, &self.anchors(), control(), None, p);
        show(label, &r); r
    }
    fn both(&self, pdf: &[u8]) { for (l, p) in presets() { self.judge(l, pdf, &p); } }
}
fn control() -> SystemTime { instant(2026, 10, 24, 0, 0, 0) }

fn m_of(pdf: &[u8]) -> String {
    let s = String::from_utf8_lossy(pdf);
    match s.rfind("/M ") { Some(i) => s[i..(i + 24).min(s.len())].replace('\n', " "), None => "no /M in file".into() }
}

fn fields(label: &str, pdf: &[u8]) {
    match nacre_pdf::Document::parse(pdf) {
        Err(e) => println!("  [nacre_pdf::Document::parse {label}] Err({e:?}) display=\"{e}\""),
        Ok(doc) => {
            println!("  [nacre_pdf::Document::parse {label}] Ok, {} signature field(s)", doc.signature_fields().len());
            for f in doc.signature_fields() { println!("    name={} id={:?} value={:?} value_unreadable={}", f.name, f.id, f.value, f.value_unreadable); }
        }
    }
}
fn v_ref(pdf: &[u8]) -> String {
    let doc = nacre_pdf::Document::parse(pdf).unwrap();
    let id = doc.signature_fields()[0].value.unwrap();
    let s = format!("{id:?}");
    let nums: Vec<String> = s.chars().map(|c| if c.is_ascii_digit() { c } else { ' ' }).collect::<String>().split_whitespace().map(String::from).collect();
    format!("/V {} {} R", nums[0], nums[1])
}

    fields("unpatched", &base);
    pki.both(&base);
    println!("QA43 signature, /V patched to `<<>>` (a direct dictionary)");
    let p = patch(&base, r.as_bytes(), b"/V <<>>");
    fields("/V <<>>", &p); pki.both(&p);
}
fn c5e() {
    let pki = Pki::new();
    let fresh = pki.fresh();
    let base = with_dss(&custom_signed(&pki.chain(&pki.signer), b"ETSI.CAdES.detached", Some(b"D:20261022080000Z")), &[], &fresh);
    let r = v_ref(&base);
    for (label, to) in [("/V null", &b"null"[..]), ("/V (x) (a string)", b"(x)"), ("/V [6] (an array)", b"[6]"), ("/V /Sig (a name)", b"/Sig"), ("/V 9 9 R (a reference to a missing object)", b"9 9 R")] {
        println!("QA43 signature, {label}");
        let mut new = b"/V ".to_vec(); new.extend_from_slice(to); let p = patch(&base, r.as_bytes(), &new);
        fields(label, &p); pki.both(&p);
    }
    println!("corpus usf-v-direct.pdf truncated to half its length");
    let f = file("attacks/usf-v-direct.pdf");
    let h = &f[..f.len() / 2];
    fields("truncated", h);
    for (l, p) in presets() { show(l, &validate_with_policy(h, &TrustAnchors::default(), SystemTime::UNIX_EPOCH, None, &p)); }
}

// ---------- c6 / c7: isolation and opened attacks over the corpus (AC6, AC7)
const SWITCHES: [&str; 9] = ["refuse_later_revisions", "refuse_sneaky_signatures", "refuse_trailing_data", "list_unusable_signatures",
    "judge_signer_key_usage", "judge_validity_before_revocation", "judge_malformed_extensions", "require_fresh_revocation", "refuse_unsupported_sub_filters"];
fn without(name: &str) -> ValidationPolicy {
    let mut p = ValidationPolicy::strict();
    match name {
        "refuse_later_revisions" => p.refuse_later_revisions = false,
        "refuse_sneaky_signatures" => p.refuse_sneaky_signatures = false,
        "refuse_trailing_data" => p.refuse_trailing_data = false,
        "list_unusable_signatures" => p.list_unusable_signatures = false,
        "judge_signer_key_usage" => p.judge_signer_key_usage = false,
        "judge_validity_before_revocation" => p.judge_validity_before_revocation = false,
        "judge_malformed_extensions" => p.judge_malformed_extensions = false,
        "require_fresh_revocation" => p.require_fresh_revocation = false,
        "refuse_unsupported_sub_filters" => p.refuse_unsupported_sub_filters = false,
        _ => unreachable!(),
    }
    p
}
struct Stub { int: Vec<u8>, int_crl: Vec<u8>, root_crl: Vec<u8> }
impl RevocationSource for Stub {
    fn revocation(&self, _c: &[u8], issuer: &[u8]) -> Result<RevocationData, nacre_core::Error> {
        Ok(RevocationData::Crl(if issuer == self.int { self.int_crl.clone() } else { self.root_crl.clone() }))
    }
}
fn verdict(r: &R) -> String {
    match r {
        Err(e) => { let s = format!("{e:?}"); format!("Err({})", s.split(|c: char| !c.is_alphanumeric()).next().unwrap()) }
        Ok(v) => format!("{:?}", v.iter().map(|s| (s.field.clone(), s.report.indication(), s.report.sub_indication())).collect::<Vec<_>>()),
    }
}
fn detail(r: &R) -> String {
    match r { Err(e) => format!("Err({e:?})"), Ok(v) => format!("{:?}", v.iter().map(|s| (&s.field, &s.defect)).collect::<Vec<_>>()) }
}
fn catalog() -> BTreeMap<String, Vec<String>> {
    let names: BTreeSet<String> = std::fs::read_dir(format!("{CORPUS}attacks")).unwrap().map(|e| e.unwrap().file_name().into_string().unwrap()).filter(|n| n.ends_with(".pdf")).collect();
    let mut cells: BTreeMap<String, Vec<String>> = BTreeMap::new();
    for doc in ["pades.md", "cms.md", "revocation.md", "tsp.md"] {
        let text = std::fs::read_to_string(format!("{CATALOG}{doc}")).unwrap();
        for line in text.lines().filter(|l| l.starts_with('|')) {
            let parts: Vec<&str> = line.trim().trim_matches('|').split('|').map(str::trim).collect();
            let Some(case) = parts.iter().find(|c| { let t = c.trim_matches('`'); names.contains(t) }) else { continue };
            let case = case.trim_matches('`').to_string();
            let last = parts.last().unwrap().to_string();
            cells.entry(case).or_default().push(format!("{doc}: {last}"));
        }
    }
    cells
}
fn ticks(cell: &str) -> Vec<String> { cell.split('`').skip(1).step_by(2).map(String::from).collect() }

fn c6(error: bool) {
    let chain_int = file("chain/intermediate.der");
    let chain_root = file("chain/root.der");
    let anchors = TrustAnchors::from_certificates([chain_root.as_slice()]).unwrap();
    use der::Decode;
    let int_c = Certificate::from_der(&chain_int).unwrap();
    let root_c = Certificate::from_der(&chain_root).unwrap();
    let t = time(2026, 10, 21, 12, 0, 0);
    let stub = Stub { int: chain_int.clone(), int_crl: crl(&int_c, &key(&[0x32; 32]), t, time(2026, 12, 1, 0, 0, 0), &[]), root_crl: crl(&root_c, &key(&[0x31; 32]), t, time(2026, 12, 1, 0, 0, 0), &[]) };
    let at = instant(2026, 10, 22, 0, 0, 0);
    let run = |pdf: &[u8], p: &ValidationPolicy| validate_with_policy(pdf, &anchors, at, Some(&stub), p);
    println!("Anchor chain/root.der, control date 2026-10-22T00:00Z, RevocationSource answering \"not revoked\" CRLs of chain intermediate and root issued 2026-10-21T12:00Z (forged by this program with keys [0x32;32] and [0x31;32])");
    if error {
        println!("Inputs that are not attacks: each must give the same verdict under strict, dss and strict minus each switch");
        let ok = file("chain/chain-dss.pdf");
        let mut inputs: Vec<(String, Vec<u8>)> = vec![("chain/chain-dss.pdf".into(), ok.clone()), ("unsigned.pdf".into(), file("unsigned.pdf")), ("empty input".into(), vec![]), ("not a PDF".into(), b"hello".to_vec())];
        let a = file("attacks/usf-v-direct.pdf"); inputs.push(("attacks/usf-v-direct.pdf truncated to 100 bytes".into(), a[..100].to_vec()));
        for (n, pdf) in inputs {
            let s = run(&pdf, &ValidationPolicy::strict());
            println!("{n}: strict {}  detail {}", verdict(&s), detail(&s));
            let dv = run(&pdf, &ValidationPolicy::dss());
            println!("  dss {}", verdict(&dv));
            let changed: Vec<&str> = SWITCHES.iter().copied().filter(|sw| verdict(&run(&pdf, &without(sw))) != verdict(&s)).collect();
            println!("  switches whose removal alone changes the verdict: {changed:?}");
        }
        return;
    }
    let cat = catalog();
    let mut files: Vec<String> = std::fs::read_dir(format!("{CORPUS}attacks")).unwrap().map(|e| e.unwrap().file_name().into_string().unwrap()).filter(|n| n.ends_with(".pdf")).collect();
    files.sort();
    let (mut mism6, mut mism7, mut not_rejected, mut defect_only) = (0, 0, 0, 0);
    for f in &files {
        let pdf = file(&format!("attacks/{f}"));
        let s = run(&pdf, &ValidationPolicy::strict());
        let dv = run(&pdf, &ValidationPolicy::dss());
        let rejected = match &s { Err(_) => true, Ok(v) => v.iter().all(|x| x.defect.is_some() || x.report.indication() != nacre_core::Indication::TotalPassed) && v.iter().any(|x| x.defect.is_some() || x.report.indication() != nacre_core::Indication::TotalPassed) || v.is_empty() };
        let strict_clean = matches!(&s, Ok(v) if v.iter().any(|x| x.defect.is_none()));
        let changed: Vec<String> = SWITCHES.iter().filter(|sw| verdict(&run(&pdf, &without(sw))) != verdict(&s)).map(|s| s.to_string()).collect();
        let dchanged: Vec<String> = SWITCHES.iter().filter(|sw| { let w = run(&pdf, &without(sw)); verdict(&w) == verdict(&s) && detail(&w) != detail(&s) }).map(|s| s.to_string()).collect();
        let opened = verdict(&dv) != verdict(&s) && matches!(&dv, Ok(v) if v.iter().any(|x| x.defect.is_none()));
        let cells = cat.get(f).cloned().unwrap_or_default();
        println!("{f}");
        println!("  strict: {}  defects {}", verdict(&s), detail(&s));
        println!("  dss:    {}  defects {}", verdict(&dv), detail(&dv));
        println!("  strict rejects (no signature listed without defect): {}", !strict_clean);
        println!("  switches whose removal alone changes the verdict: {changed:?}");
        if !dchanged.is_empty() { defect_only += 1; println!("  switches changing only the defect (same verdict): {dchanged:?}"); }
        println!("  opened under dss: {opened}");
        println!("  catalog rows: {}", cells.len());
        for c in &cells { println!("    {c}"); }
        let named: BTreeSet<String> = cells.iter().flat_map(|c| ticks(c.split_once(": ").unwrap().1)).filter(|n| n != "dss()").collect();
        let starts = cells.iter().any(|c| c.split_once(": ").unwrap().1.starts_with("`dss()`"));
        let got: BTreeSet<String> = changed.iter().cloned().collect();
        if cells.len() != 1 || named != got { mism6 += 1; println!("  MISMATCH AC6: catalog names {named:?}, observed {got:?}, rows {}", cells.len()); }
        if starts != opened { mism7 += 1; println!("  MISMATCH AC7: catalog opened={starts}, observed opened={opened}"); }
        if strict_clean { not_rejected += 1; println!("  NOT REJECTED UNDER strict"); }
        let _ = rejected;
    }
    let orphan: Vec<&String> = cat.keys().filter(|k| !files.contains(k)).collect();
    println!("SUMMARY files={} AC6 mismatches={mism6} AC7 mismatches={mism7} not rejected under strict={not_rejected} files with defect-only changes={defect_only} catalog cases with no file={orphan:?}", files.len());
}
