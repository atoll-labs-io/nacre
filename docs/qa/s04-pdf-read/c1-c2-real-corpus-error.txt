use std::time::Instant;

fn keys(d: &nacre_pdf::Dictionary) -> String {
    d.iter().map(|(k, _)| format!("/{}", String::from_utf8_lossy(k))).collect::<Vec<_>>().join(" ")
}

fn main() {
    let path = match std::env::args().nth(1) {
        Some(p) => p,
        None => { eprintln!("usage: probe <file.pdf>"); std::process::exit(2) }
    };
    let bytes = match std::fs::read(&path) {
        Ok(b) => b,
        Err(e) => { eprintln!("cannot read {path}: {e}"); std::process::exit(2) }
    };
    println!("file: {path} ({} bytes)", bytes.len());
    let start = Instant::now();
    let result = std::panic::catch_unwind(|| match nacre_pdf::Document::parse(&bytes) {
        Ok(doc) => {
            for (i, r) in doc.revisions().iter().enumerate() {
                println!("revision {i}: end={} startxref={} xref={:?}", r.end, r.startxref, r.xref);
            }
            println!("trailer: {}", keys(doc.trailer()));
            println!("catalog {}: {}", doc.catalog_id(), keys(doc.catalog()));
            println!("signature fields: {}", doc.signature_fields().len());
            for f in doc.signature_fields() {
                let v = f.value.map(|v| v.to_string()).unwrap_or_else(|| "empty".into());
                println!("  field {:?} {} /V {}", f.name, f.id, v);
            }
            if let Ok(n) = std::env::var("PROBE_OBJECT") {
                let id = nacre_pdf::ObjectId::new(n.parse().unwrap_or(0), 0);
                match doc.object(id) {
                    Ok(o) => println!("object {id}: {}", format!("{o:?}").chars().take(160).collect::<String>()),
                    Err(e) => { println!("object {id}: error: {e}"); return 1; }
                }
            }
            0
        }
        Err(e) => { println!("error: {e}"); println!("debug: {e:?}"); 1 }
    });
    println!("elapsed: {:?}", start.elapsed());
    if let Ok(s) = std::fs::read_to_string("/proc/self/status") { for l in s.lines().filter(|l| l.starts_with("VmHWM")) { println!("peak memory: {}", l.trim_start_matches("VmHWM:").trim()); } }
    match result {
        Ok(code) => std::process::exit(code),
        Err(_) => { println!("PANIC"); std::process::exit(101) }
    }
}
