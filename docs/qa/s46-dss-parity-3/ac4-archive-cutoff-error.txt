/// Re-sign a forged OCSP response with `key` after `f` altered its tbsResponseData.
fn mutate(resp: &[u8], key: &SigningKey, f: impl FnOnce(&mut ocspasn1::ResponseData)) -> Vec<u8> {
    let mut outer = ocspasn1::OcspResponse::from_der(resp).unwrap();
    let rb = outer.response_bytes.as_mut().unwrap();
    let mut basic = ocspasn1::BasicOcspResponse::from_der(rb.response.as_bytes()).unwrap();
    f(&mut basic.tbs_response_data);
    let sig: DerSignature = key.sign_prehash(&Sha256::digest(basic.tbs_response_data.to_der().unwrap())).unwrap();
    basic.signature = BitString::from_bytes(sig.as_bytes()).unwrap();
    rb.response = OctetString::new(basic.to_der().unwrap()).unwrap();
    outer.to_der().unwrap()
}

fn read(path: &str) -> Vec<u8> { std::fs::read(path).unwrap_or_else(|e| panic!("{path}: {e}")) }
fn verdict(r: &R) -> String {
    match r {
        Err(e) => { let s = format!("{e:?}"); format!("Err({})", s.split(|c: char| !c.is_alphanumeric()).next().unwrap()) }
        Ok(v) => format!("{:?}", v.iter().map(|s| (s.field.clone(), s.report.indication(), s.report.sub_indication())).collect::<Vec<_>>()),
    }
}
fn detail(r: &R) -> String { match r { Err(e) => format!("Err({e:?})"), Ok(v) => format!("{:?}", v.iter().map(|s| (&s.field, &s.defect)).collect::<Vec<_>>()) } }
fn catalog(names: &BTreeSet<String>) -> BTreeMap<String, Vec<String>> {
    let mut cells: BTreeMap<String, Vec<String>> = BTreeMap::new();
    for doc in ["pades.md", "cms.md", "revocation.md", "tsp.md"] {
        let Ok(text) = std::fs::read_to_string(format!("{CATALOG}{doc}")) else { continue };
        for line in text.lines().filter(|l| l.starts_with('|')) {
            let parts: Vec<&str> = line.trim().trim_matches('|').split('|').map(str::trim).collect();
            let Some(case) = parts.iter().find(|c| names.contains(c.trim_matches('`'))) else { continue };
            cells.entry(case.trim_matches('`').to_string()).or_default().push(format!("{doc}: {}", parts.last().unwrap()));
        }
    }
    cells
}

        println!("{f}: strict {} | dss {} | switches changing it {changed:?} | catalogue names {named:?} | opened {opened}", verdict(&s), verdict(&dv));
        if cells.len() != 1 || named != changed { m6 += 1; println!("  MISMATCH isolation: rows {}", cells.len()); }
        if starts != opened { m7 += 1; println!("  MISMATCH opened: catalogue {starts}, observed {opened}"); }
    }
    println!("SUMMARY files={} isolation mismatches={m6} opened mismatches={m7}", files.len());
    for doc in ["pades.md", "cms.md", "revocation.md", "tsp.md"] {
        let text = std::fs::read_to_string(format!("{CATALOG}{doc}")).unwrap_or_default();
        for sw in ["judge_issuer_of_revoked_data", "ignore_expired_certs_on_crl", "archiveCutoff", "ADR-0030"] { println!("  {doc} mentions {sw}: {}", text.matches(sw).count()); }
    }
}
