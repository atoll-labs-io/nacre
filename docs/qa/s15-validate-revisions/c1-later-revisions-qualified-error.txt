use std::process::ExitCode;

fn main() -> ExitCode {
    let Some(path) = std::env::args().nth(1) else {
        eprintln!("usage: qa-scen <file.pdf>");
        return ExitCode::from(2);
    };
    let pdf = match std::fs::read(&path) {
        Ok(b) => b,
        Err(e) => {
            eprintln!("read error: {e}");
            return ExitCode::from(2);
        }
    };
    println!("file: {path} ({} bytes)", pdf.len());
    match nacre_validate::validate(&pdf) {
        Ok(sigs) => {
            println!("signatures: {}", sigs.len());
            for s in &sigs {
                println!("- field {:?}", s.field);
                println!("  indication: {} / {:?}", s.report.indication().as_str(),
                    s.report.sub_indication().map(|x| x.as_str()));
                println!("  digest_verified={} signature_verified={}", s.digest_verified, s.signature_verified);
                println!("  defect: {:?}", s.defect);
                println!("  later_revisions: {}", s.later_revisions.len());
                for r in &s.later_revisions {
                    println!("    end={} changed={:?}", r.end, r.changed);
                    println!("      verdict={:?}", r.verdict);
                }
            }
            ExitCode::SUCCESS
        }
        Err(e) => {
            println!("error: {e} (kind {:?})", e.kind());
            println!("debug: {e:?}");
            ExitCode::from(1)
        }
    }
}

#!/bin/sh
# usage: append.sh in out count size root
in=$1; out=$2; n=$3; size=$4; root=$5
cp "$in" "$out"
prev=$(tail -c 64 "$in" | grep -a -A1 startxref | tail -1)
i=0
while [ $i -lt $n ]; do
  off=$(stat -c %s "$out")
  printf '\nxref\n0 0\ntrailer\n<< /Size %s /Prev %s /Root %s >>\nstartxref\n%s\n%%%%EOF\n' "$size" "$prev" "$root" "$((off+1))" >> "$out"
  prev=$((off+1)); i=$((i+1))
done

#!/bin/sh
# usage: hostile-refs.sh in out nrefs
# Appends one revision redefining the catalog (1 0) with /Perms /DocMDP 20 0 R, where
# 20 0 is a signature dictionary whose /Reference array holds nrefs DocMDP entries,
# then one empty xref-only revision judged under that hostile state.
in=$1; out=$2; n=$3
cp "$in" "$out"
prev=$(tail -c 64 "$in" | grep -a -A1 startxref | tail -1)
printf '\n' >> "$out"
o1=$(stat -c %s "$out")
printf '1 0 obj\n<< /Type /Catalog /Pages 2 0 R /AcroForm 6 0 R /Perms << /DocMDP 20 0 R >> >>\nendobj\n' >> "$out"
o20=$(stat -c %s "$out")
refs=""; i=0
while [ $i -lt $n ]; do refs="$refs<< /Type /SigRef /TransformMethod /DocMDP /TransformParams << /P 2 >> >> "; i=$((i+1)); done
printf '20 0 obj\n<< /Type /Sig /Reference [ %s] >>\nendobj\n' "$refs" >> "$out"
x=$(stat -c %s "$out")
printf 'xref\n1 1\n%010d 00000 n \n20 1\n%010d 00000 n \ntrailer\n<< /Size 21 /Prev %s /Root 1 0 R >>\nstartxref\n%s\n%%%%EOF\n' "$o1" "$o20" "$prev" "$x" >> "$out"
off=$(stat -c %s "$out")
printf '\nxref\n0 0\ntrailer\n<< /Size 21 /Prev %s /Root 1 0 R >>\nstartxref\n%s\n%%%%EOF\n' "$x" "$((off+1))" >> "$out"
