use std::process::ExitCode;

fn main() -> ExitCode {
    let Some(path) = std::env::args().nth(1) else {
        eprintln!("usage: qa-scen <file.pdf>");
        return ExitCode::from(2);
    };
    let pdf = match std::fs::read(&path) {
        Ok(b) => b,
        Err(e) => {
            eprintln!("read error: {e}");
            return ExitCode::from(2);
        }
    };
    println!("file: {path} ({} bytes)", pdf.len());
    match nacre_validate::validate(&pdf) {
        Ok(sigs) => {
            println!("signatures: {}", sigs.len());
            for s in &sigs {
                println!("- field {:?}", s.field);
                println!("  indication: {} / {:?}", s.report.indication().as_str(),
                    s.report.sub_indication().map(|x| x.as_str()));
                println!("  digest_verified={} signature_verified={}", s.digest_verified, s.signature_verified);
                println!("  defect: {:?}", s.defect);
                println!("  later_revisions: {}", s.later_revisions.len());
                for r in &s.later_revisions {
                    println!("    end={} changed={:?}", r.end, r.changed);
                    println!("      verdict={:?}", r.verdict);
                }
            }
            ExitCode::SUCCESS
        }
        Err(e) => {
            println!("error: {e} (kind {:?})", e.kind());
            println!("debug: {e:?}");
            ExitCode::from(1)
        }
    }
}
