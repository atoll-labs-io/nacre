fn show(label: &str, r: &Result<Vec<SignatureValidation>, Error>) {
    match r {
        Err(e) => println!("  [{label}] Err({e:?}) kind={:?} display=\"{e}\"", e.kind()),
        Ok(v) => {
            println!("  [{label}] signatures: {}", v.len());
            for s in v {
                println!(
                    "    field={} indication={:?} sub={:?} defect={:?} digest={} sig={} later_revisions={} chain={:?} bst={:?}",
                    s.field, s.report.indication(), s.report.sub_indication(), s.defect,
                    s.digest_verified, s.signature_verified, s.later_revisions.len(),
                    s.chain.as_ref().map(|c| c.certificates.len()), s.best_signature_time
                );
            }
        }
    }
}

fn file(name: &str) -> Vec<u8> {
    std::fs::read(format!("{CORPUS}{name}")).unwrap_or_else(|e| panic!("{name}: {e}"))
}

fn d(c: &Certificate) -> Vec<u8> { c.to_der().unwrap() }

struct Pki { root: Certificate, rk: p256::ecdsa::SigningKey, int: Certificate, ik: p256::ecdsa::SigningKey, signer: Certificate, sk: p256::ecdsa::SigningKey }
impl Pki {
    fn new() -> Self {
        let rk = key(&[0x51; 32]); let ik = key(&[0x52; 32]); let sk = key(&[0x53; 32]);
        let root = issue(Role::Ca, "CN=QA root,O=QA,C=FR", 1, &rk, None, &rk);
        let int = issue(Role::Ca, "CN=QA intermediate,O=QA,C=FR", 2, &ik, Some(&root), &rk);
        let signer = issue(Role::Signer, "CN=QA signer,O=QA,C=FR", 3, &sk, Some(&int), &ik);
        Pki { root, rk, int, ik, signer, sk }
    }
    fn signer_valid(&self, nb: x509_cert::time::Time, na: x509_cert::time::Time) -> Certificate {
        issue_valid(Role::Signer, "CN=QA signer 2,O=QA,C=FR", 9, &self.sk, Some(&self.int), &self.ik, nb, na)
    }
    fn sign_with(&self, cert: &Certificate, m: der::DateTime) -> Vec<u8> {
        let s = ChainSigner { key: self.sk.clone(), chain: vec![d(cert), d(&self.int)] };
        nacre_pades::sign_at_level(UNSIGNED, &s, m, Level::B).unwrap()
    }
    fn crl_int(&self, at: x509_cert::time::Time, rev: &[(&Certificate, x509_cert::time::Time, Option<CrlReason>)]) -> Vec<u8> {
        crl_with_reasons(&self.int, &self.ik, at, time(2026, 12, 1, 0, 0, 0), rev)
    }
    fn crl_root(&self, at: x509_cert::time::Time) -> Vec<u8> {
        crl(&self.root, &self.rk, at, time(2026, 12, 1, 0, 0, 0), &[])
    }
    fn judge(&self, pdf: &[u8], at: SystemTime, p: &ValidationPolicy) -> Result<Vec<SignatureValidation>, Error> {
        let a = TrustAnchors::from_certificates([d(&self.root).as_slice()]).unwrap();
        validate_with_policy(pdf, &a, at, None, p)
    }
}

fn off(f: impl FnOnce(&mut ValidationPolicy)) -> ValidationPolicy { let mut p = ValidationPolicy::strict(); f(&mut p); p }

            let late = pki.crl_int(time(2026,10,21,23,0,0), &[(&pki.signer, time(2026,10,23,5,0,0), Some(CrlReason::KeyCompromise))]);
            let pdf = with_dss(&base, &[], &[late, pki.crl_root(time(2026,10,22,10,0,0))]);

            let edge = pki.crl_int(time(2026,10,21,23,0,0), &[(&pki.signer, time(2026,10,23,0,0,0), Some(CrlReason::KeyCompromise))]);

            let other = issue(Role::Signer, "CN=QA other,O=QA,C=FR", 77, &pki.sk, Some(&pki.int), &pki.ik);
            let o = pki.crl_int(time(2026,10,21,23,0,0), &[(&other, time(2026,10,21,22,0,0), Some(CrlReason::KeyCompromise))]);

            show("strict", &pki.judge(&pdf, at, &ValidationPolicy::strict()));
            println!("same old (25 h) CRL with a future-dated entry, under strict() with require_fresh_revocation off and under dss(), for comparison");
            let late = pki.crl_int(time(2026,10,21,23,0,0), &[(&pki.signer, time(2026,10,23,5,0,0), Some(CrlReason::KeyCompromise))]);
            let pdf = with_dss(&base, &[], &[late, pki.crl_root(time(2026,10,22,10,0,0))]);

            show("dss", &pki.judge(&pdf, at, &ValidationPolicy::dss()));
            println!("a CRL stream of garbage bytes in /DSS");
