```rust
// Criterion 1: an imprint gives a DER TimeStampReq with a nonce and certReq; the
// TimeStampResp is read and its TSTInfo verified.
use nacre_core::DigestAlgorithm;
use nacre_tsp::{Request, verify_response};
use sha2::{Digest, Sha256};
fn main() {
    let dir = std::env::args().nth(1).unwrap();
    let data = std::fs::read(format!("{dir}/data.bin")).unwrap();
    let imprint = Sha256::digest(&data);
    let nonce = [0xd4, 0x1c, 0xbf, 0xf4, 0xfd, 0xed, 0x9d, 0x81];
    let req = Request::new(DigestAlgorithm::Sha256, &imprint, &nonce).unwrap();
    let der = req.to_der().unwrap();
    std::fs::write(std::env::args().nth(2).unwrap(), &der).unwrap();
    let openssl = std::fs::read(format!("{dir}/request.tsq")).unwrap();
    println!("request DER: {} bytes, equal to openssl's request.tsq: {}", der.len(), der == openssl);
    let random = Request::with_random_nonce(DigestAlgorithm::Sha256, &imprint).unwrap();
    println!("random nonce request: nonce {} bytes", random.nonce().len());
    for name in ["bench.tsr", "digicert.tsr"] {
        let resp = std::fs::read(format!("{dir}/{name}")).unwrap();
        match verify_response(&resp, &req) {
            Ok(t) => println!("{name}: verified, genTime {}, serial {} bytes, nonce equal {}, token {} bytes, TSA cert {} bytes",
                t.gen_time(), t.serial_number().len(), t.nonce() == Some(&nonce[..]), t.token().len(), t.tsa_certificate().len()),
            Err(e) => println!("{name}: ERROR {e} (kind {:?})", e.kind()),
        }
    }
}
```
