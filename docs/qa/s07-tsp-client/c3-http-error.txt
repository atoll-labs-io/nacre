## Script (src/bin/c3_http_error.rs of a scratch crate depending on the copied crates/nacre-tsp, features http)
```rust
// Criterion 3, invalid input: nothing listens, a server that is not a TSA, a malformed
// URL, an imprint of the wrong length.
use nacre_core::{DigestAlgorithm, TimestampAuthority};
use nacre_tsp::HttpTsa;
fn main() {
    let args: Vec<String> = std::env::args().skip(1).collect();
    for (label, url, len) in [
        ("closed port", args[0].as_str(), 32usize),
        ("not a TSA (python http.server, no POST handler)", args[1].as_str(), 32),
        ("malformed URL", "not a url", 32),
        ("imprint of 5 bytes", args[2].as_str(), 5),
    ] {
        match HttpTsa::new(url).timestamp(DigestAlgorithm::Sha256, &vec![7u8; len]) {
            Ok(t) => println!("{label}: TOKEN {} bytes", t.len()),
            Err(e) => {
                let src = std::error::Error::source(&e).map(|s| s.to_string());
                println!("{label}: {e} | debug {e:?} | source: {src:?} | kind {:?}", e.kind());
            }
        }
    }
}
```
