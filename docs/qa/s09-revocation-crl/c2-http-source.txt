use nacre_core::{RevocationData, RevocationSource};
use nacre_revocation::{Crl, HttpCrlSource};
use qa_s09::*;
use std::time::SystemTime;

fn fetch(label: &str, source: &dyn RevocationSource, leaf: &[u8], issuer: &[u8], at: SystemTime) {
    match source.revocation(leaf, issuer) {
        Ok(RevocationData::Crl(bytes)) => {
            let status = Crl::from_der(&bytes).and_then(|c| c.status(leaf, issuer, at));
            match status {
                Ok(s) => println!("{label}: downloaded {} bytes; status at {} = {:?}", bytes.len(), iso(at), s),
                Err(e) => println!("{label}: downloaded {} bytes; status Err({})", bytes.len(), show(&e)),
            }
        }
        Ok(other) => println!("{label}: unexpected {other:?}"),
        Err(e) => {
            let inner = std::error::Error::source(&e).and_then(|s| s.downcast_ref::<nacre_revocation::Error>());
            println!(
                "{label}: Err(core kind {:?}; chain: {}; inner: {})",
                e.kind(),
                chain(&e),
                inner.map(|i| format!("{:?} kind {:?}", i, i.kind())).unwrap_or("none".into())
            );
        }
    }
}

fn main() {
    let mode = std::env::args().nth(1).unwrap_or_default();
    let at = date(2027, 1, 1);
    let lo = HttpCrlSource::allowing_loopback();
    let public = HttpCrlSource::new();
    match mode.as_str() {
        "ok" => {
            for leaf in ["ec-leaf", "ec-revoked"] {
                println!("distribution_point({leaf}) = {:?}", HttpCrlSource::distribution_point(&read(&format!("{leaf}.der"))));
                fetch(&format!("allowing_loopback, {leaf}"), &lo, &read(&format!("{leaf}.der")), &read("ec-ca.der"), at);
            }
            for leaf in ["rsa-leaf", "rsa-revoked"] {
                fetch(&format!("allowing_loopback, {leaf}"), &lo, &read(&format!("{leaf}.der")), &read("rsa-ca.der"), at);
            }
        }
        "s02" => {
            let at = date(2030, 1, 1);
            for leaf in ["s02-revoked", "s02-signer"] {
                println!("distribution_point({leaf}) = {:?}", HttpCrlSource::distribution_point(&read(&format!("{leaf}.der"))));
                fetch(&format!("allowing_loopback, {leaf}"), &lo, &read(&format!("{leaf}.der")), &read("s02-intermediate.der"), at);
            }
        }
        "refused" => {
            for leaf in ["ec-leaf", "dp-localhost", "dp-ipv6-loop", "dp-mapped", "dp-decimal", "dp-zero", "dp-private", "dp-metadata", "dp-unresolvable"] {
                println!("distribution_point({leaf}) = {:?}", HttpCrlSource::distribution_point(&read(&format!("{leaf}.der"))));
                fetch(&format!("new(), {leaf}"), &public, &read(&format!("{leaf}.der")), &read("ec-ca.der"), at);
            }
            for leaf in ["dp-localhost", "dp-ipv6-loop", "dp-mapped", "dp-decimal", "dp-zero", "dp-private", "dp-metadata"] {
                fetch(&format!("allowing_loopback(), {leaf}"), &lo, &read(&format!("{leaf}.der")), &read("ec-ca.der"), at);
            }
        }
        "error" => {
            for leaf in ["dp-redirect", "dp-404", "dp-junk", "dp-cert", "dp-down", "dp-https", "dp-ldap", "dp-nocdp", "p384-revoked"] {
                println!("distribution_point({leaf}) = {:?}", HttpCrlSource::distribution_point(&read(&format!("{leaf}.der"))));
                let issuer = if leaf.starts_with("p384") { read("p384-ca.der") } else { read("ec-ca.der") };
                fetch(&format!("allowing_loopback, {leaf}"), &lo, &read(&format!("{leaf}.der")), &issuer, at);
            }
            fetch("allowing_loopback, a CRL given as the certificate", &lo, &read("ec.crl"), &read("ec-ca.der"), at);
            fetch("allowing_loopback, empty certificate", &lo, &[], &read("ec-ca.der"), at);
        }
        "proxy" => {
            for v in ["HTTP_PROXY", "http_proxy", "ALL_PROXY", "ALL_PROXY"] {
                println!("{v}={}", std::env::var(v).unwrap_or_default());
            }
            fetch("allowing_loopback under a proxy environment, ec-revoked", &lo, &read("ec-revoked.der"), &read("ec-ca.der"), at);
            fetch("new() under a proxy environment, dp-private", &public, &read("dp-private.der"), &read("ec-ca.der"), at);
        }
        _ => unreachable!(),
    }
}
