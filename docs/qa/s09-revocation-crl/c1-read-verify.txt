use nacre_revocation::Crl;
use qa_s09::*;

fn describe(label: &str, crl: &[u8], ca: &[u8]) {
    println!("== {label}");
    match Crl::from_der(crl) {
        Err(e) => println!("from_der: Err({})", show(&e)),
        Ok(c) => {
            println!("issuer:      {}", c.issuer());
            println!("thisUpdate:  {}", iso(c.this_update()));
            println!("nextUpdate:  {}", c.next_update().map(iso).unwrap_or("none".into()));
            match c.revoked() {
                Ok(list) => {
                    println!("revoked:     {} entr{}", list.len(), if list.len() == 1 { "y" } else { "ies" });
                    for r in list {
                        println!("  serial {} date {} reason {:?}", hex(&r.serial), iso(r.date), r.reason);
                    }
                }
                Err(e) => println!("revoked: Err({})", show(&e)),
            }
            match c.verify_signature(ca) {
                Ok(()) => println!("verify_signature: Ok"),
                Err(e) => println!("verify_signature: Err({})", show(&e)),
            }
        }
    }
}

fn verify_only(label: &str, crl: &[u8], ca: &[u8]) {
    match Crl::from_der(crl) {
        Err(e) => println!("{label}: from_der Err({})", show(&e)),
        Ok(c) => match c.verify_signature(ca) {
            Ok(()) => println!("{label}: verify_signature Ok"),
            Err(e) => println!("{label}: verify_signature Err({})", show(&e)),
        },
    }
}

fn main() {
    let mode = std::env::args().nth(1).unwrap_or_default();
    if mode != "error" {
        describe("committed fixture openssl/ec-ca.crl, issuer openssl/ec-ca.der", &corpus("openssl/ec-ca.crl"), &corpus("openssl/ec-ca.der"));
        describe("committed fixture openssl/rsa-ca.crl, issuer openssl/rsa-ca.der", &corpus("openssl/rsa-ca.crl"), &corpus("openssl/rsa-ca.der"));
        describe("walk's OpenSSL P-256 CA: ec.crl, ec-ca.der", &read("ec.crl"), &read("ec-ca.der"));
        describe("walk's OpenSSL RSA-2048 CA: rsa.crl, rsa-ca.der", &read("rsa.crl"), &read("rsa-ca.der"));
        describe("walk's second P-256 CA (empty CRL): ec2.crl, ec2-ca.der", &read("ec2.crl"), &read("ec2-ca.der"));
        describe("s02 PKI: intermediate.crl, intermediate cert", &read("s02-intermediate.crl"), &read("s02-intermediate.der"));
        describe("s02 PKI: root.crl, root cert", &read("s02-root.crl"), &read("s02-root.der"));
        return;
    }
    let ec = read("ec.crl");
    verify_only("P-256 CRL checked with the RSA CA", &ec, &read("rsa-ca.der"));
    verify_only("P-256 CRL checked with another P-256 CA", &ec, &read("ec2-ca.der"));
    verify_only("P-256 CRL checked with the s02 root", &ec, &read("s02-root.der"));
    verify_only("P-256 CRL checked with its own leaf (not a CRL signer)", &ec, &read("ec-leaf.der"));
    let mut flipped = ec.clone();
    // A byte of thisUpdate: find "261001" UTCTime digits.
    let pos = flipped.windows(6).position(|w| w == b"261001").unwrap();
    flipped[pos + 5] = b'2';
    verify_only("P-256 CRL with thisUpdate 261001 -> 261002 after signing", &flipped, &read("ec-ca.der"));
    describe("P-384 CA: p384.crl, p384-ca.der", &read("p384.crl"), &read("p384-ca.der"));
    let pem = std::process::Command::new("openssl").args(["crl", "-inform", "DER", "-in", &format!("{PKI}/ec.crl")]).output().unwrap().stdout;
    verify_only("the same CRL as PEM text", &pem, &read("ec-ca.der"));
    verify_only("empty input", &[], &read("ec-ca.der"));
    verify_only("a certificate given as the CRL", &read("ec-leaf.der"), &read("ec-ca.der"));
    let mut trailing = ec.clone();
    trailing.extend_from_slice(&[0, 0]);
    verify_only("the CRL followed by two zero bytes", &trailing, &read("ec-ca.der"));
    let random: Vec<u8> = (0u32..300).map(|i| (i.wrapping_mul(2654435761) >> 13) as u8).collect();
    verify_only("300 pseudo-random bytes", &random, &read("ec-ca.der"));
    verify_only("issuer argument: the CRL itself instead of a certificate", &ec, &ec);
    verify_only("issuer argument: empty", &ec, &[]);
}
