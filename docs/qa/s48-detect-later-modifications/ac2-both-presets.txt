fn at() -> SystemTime {
    x509_cert::time::Time::from(der::DateTime::new(2026, 10, 23, 0, 0, 0).unwrap()).to_system_time()
}

fn judge(name: &str, pdf: &[u8], anchors: &TrustAnchors) {
    for (label, policy) in [("strict", ValidationPolicy::strict()), ("dss", ValidationPolicy::dss())] {
        match validate_with_policy(pdf, anchors, at(), None, &policy) {
            Ok(results) if results.is_empty() => println!("{name} [{label}] no signature"),
            Ok(results) => {
                for r in results {
                    println!(
                        "{name} [{label}] {} -> {:?} / {:?}; defect={:?}; digest={} sig={}; later_revisions={}",
                        r.field, r.report.indication(), r.report.sub_indication(), r.defect,
                        r.digest_verified, r.signature_verified, r.later_revisions.len()
                    );
                }
            }
            Err(e) => println!("{name} [{label}] Err({e:?}) kind={:?}", e.kind()),
        }
    }
}

fn dict_of(doc: &Document<'_>, id: ObjectId) -> nacre_pdf::Dictionary {
    let Object::Dictionary(d) = doc.object(id).unwrap() else { panic!("not a dictionary") };
    d
}

fn without(d: &nacre_pdf::Dictionary, key: &[u8]) -> nacre_pdf::Dictionary {
    let mut out = nacre_pdf::Dictionary::new();
    for (k, v) in d.iter() {
        if k != key { out.insert(k, v.clone()); }
    }
    out
}
