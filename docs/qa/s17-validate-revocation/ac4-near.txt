fn der(c: &Certificate) -> Vec<u8> { c.to_der().unwrap() }

struct Pki { root: Certificate, rk: SigningKey, int: Certificate, ik: SigningKey,
             signer: Certificate, sk: SigningKey, tsa: Certificate, tk: SigningKey }

impl Pki {
    fn new() -> Self {
        let rk = key(&[0x71; 32]); let ik = key(&[0x72; 32]);
        let sk = key(&[0x73; 32]); let tk = key(&[0x74; 32]);
        let root = issue(Role::Ca, "CN=QA root,O=QA,C=FR", 11, &rk, None, &rk);
        let int = issue(Role::Ca, "CN=QA intermediate,O=QA,C=FR", 12, &ik, Some(&root), &rk);
        let signer = issue(Role::Signer, "CN=QA signer,O=QA,C=FR", 13, &sk, Some(&int), &ik);
        let tsa = issue(Role::Tsa, "CN=QA TSA,O=QA,C=FR", 14, &tk, Some(&int), &ik);
        Self { root, rk, int, ik, signer, sk, tsa, tk }
    }
    fn chain_signer(&self) -> ChainSigner { ChainSigner { key: self.sk.clone(), chain: vec![der(&self.signer), der(&self.int)] } }
    fn tsa_at(&self, h: u8, m: u8) -> ForgedTsa { ForgedTsa { key: self.tk.clone(), certificates: vec![der(&self.tsa)], gen_time: DateTime::new(2026, 10, 22, h, m, 0).unwrap(), imprint: None } }
    fn int_crl(&self, this_update: Time, revoked: &[(&Certificate, Time)]) -> Vec<u8> { crl(&self.int, &self.ik, this_update, time(2026, 12, 1, 0, 0, 0), revoked) }
    fn root_crl_at(&self, this_update: Time, revoked: &[(&Certificate, Time)]) -> Vec<u8> { crl(&self.root, &self.rk, this_update, time(2026, 12, 1, 0, 0, 0), revoked) }
    fn root_crl(&self, revoked: &[(&Certificate, Time)]) -> Vec<u8> { crl(&self.root, &self.rk, time(2026, 10, 22, 14, 0, 0), time(2026, 12, 1, 0, 0, 0), revoked) }
    fn anchors(&self) -> TrustAnchors { let r = der(&self.root); TrustAnchors::from_certificates([r.as_slice()]).unwrap() }
    fn certs(&self) -> Vec<Vec<u8>> { vec![der(&self.int), der(&self.root)] }
}

/// Answers from fixed CRLs, by issuer; counts nothing, keeps nothing.
struct StubSource { pki_int: Vec<u8>, int_crl: Vec<u8>, root_crl: Vec<u8> }
impl RevocationSource for StubSource {
    fn revocation(&self, _certificate: &[u8], issuer: &[u8]) -> Result<RevocationData, nacre_core::Error> {
        if issuer == self.pki_int.as_slice() { Ok(RevocationData::Crl(self.int_crl.clone())) } else { Ok(RevocationData::Crl(self.root_crl.clone())) }
    }
}
struct Failing;
impl RevocationSource for Failing {
    fn revocation(&self, _: &[u8], _: &[u8]) -> Result<RevocationData, nacre_core::Error> {
        Err(nacre_core::Error::Signer("network unreachable (stub)".into()))
    }
}
struct Junk;
impl RevocationSource for Junk {
    fn revocation(&self, _: &[u8], _: &[u8]) -> Result<RevocationData, nacre_core::Error> {
        Ok(RevocationData::Crl(b"not a CRL at all".to_vec()))
    }
}

fn fmt_time(t: Option<SystemTime>) -> String {
    t.map(|t| { let x = Time::try_from(t).map(|x| x.to_string()).unwrap_or_else(|_| format!("{t:?}")); x }).unwrap_or_else(|| "None".into())
}

fn show(label: &str, r: Result<Vec<nacre_validate::SignatureValidation>, nacre_validate::Error>) {
    println!("== {label}");
    match r {
        Err(e) => println!("   Err: {e}  (kind {:?})", e.kind()),
        Ok(v) => for s in v {
            println!("   field={} indication={} sub={:?} level={:?}", s.field, s.report.indication().as_str(), s.report.sub_indication(), s.report.level());
            println!("   digest_verified={} signature_verified={} chain_len={:?}", s.digest_verified, s.signature_verified, s.chain.as_ref().map(|c| c.certificates.len()));
            println!("   defect={:?}", s.defect.as_ref().map(|d| format!("{d:?}")));
            println!("   best_signature_time={}", fmt_time(s.best_signature_time));
        }
    }
}

            let forged = crl(&p.int, &key(&[0x7e; 32]), time(2026, 10, 24, 18, 0, 0), time(2026, 12, 1, 0, 0, 0), &[]);
            show("B-B, fresh /DSS CRL (10-24T18:00) claiming the intermediate but signed by another key, listing nothing", validate_with_anchors(&with_dss(&bb, &p.certs(), &[forged, p.root_crl_at(time(2026, 10, 24, 18, 0, 0), &[])]), &p.anchors(), control));
            show("B-B, /DSS CRLs dated 10-26 (after the control date), nothing revoked", validate_with_anchors(&with_dss(&bb, &p.certs(), &[p.int_crl(time(2026, 10, 26, 0, 0, 0), &[]), p.root_crl_at(time(2026, 10, 26, 0, 0, 0), &[])]), &p.anchors(), control));
        }
        "ac4-near" => {
            // AC4 and its invalid input with the control date of the repo's tests (10 h after the CRLs).
            let near = instant(2026, 10, 23, 0, 0, 0);
            let pdf = signed(&p.chain_signer(), Level::T { tsa: &p.tsa_at(8, 0) });
            let tsa_rev = with_dss(&pdf, &p.certs(), &[p.int_crl(time(2026, 10, 22, 14, 0, 0), &[(&p.signer, time(2026, 10, 22, 12, 0, 0)), (&p.tsa, time(2026, 10, 22, 13, 0, 0))]), p.root_crl(&[])]);
            show("control 10-23T00:00: B-T 08:00, signer revoked 12:00, TSA revoked 13:00", validate_with_anchors(&tsa_rev, &p.anchors(), near));
            let tsa = ForgedTsa { key: p.tk.clone(), certificates: vec![der(&p.tsa)], gen_time: DateTime::new(2026, 10, 22, 8, 0, 0).unwrap(), imprint: Some(vec![0u8; 32]) };
            let f = with_dss(&signed(&p.chain_signer(), Level::T { tsa: &tsa }), &p.certs(), &[p.int_crl(time(2026, 10, 22, 14, 0, 0), &[(&p.signer, time(2026, 10, 22, 12, 0, 0))]), p.root_crl(&[])]);
            show("control 10-23T00:00: B-T token over another imprint, signer revoked 12:00", validate_with_anchors(&f, &p.anchors(), near));
            let rev = with_dss(&signed(&p.chain_signer(), Level::B), &p.certs(), &[p.int_crl(time(2026, 10, 22, 14, 0, 0), &[(&p.signer, time(2026, 10, 22, 12, 0, 0))]), p.root_crl(&[])]);
            show("control 10-23T00:00: B-B, signer revoked 12:00 (AC2 with data 10 h old)", validate_with_anchors(&rev, &p.anchors(), near));
        }
        _ => eprintln!("usage: qa-s17 <ac1|ac1-error|ac2|ac2-error|ac3|ac3-error|ac4|ac4-error>"),
    }
