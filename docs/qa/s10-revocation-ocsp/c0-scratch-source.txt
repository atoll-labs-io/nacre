use std::time::{Duration, SystemTime, UNIX_EPOCH};
use nacre_core::{RevocationData, RevocationSource};
use nacre_revocation::{Error, Fallback, HttpCrlSource, HttpOcspSource, OcspRequest, OcspResponse};

fn hex(b: &[u8]) -> String { b.iter().map(|x| format!("{x:02x}")).collect() }
fn unhex(s: &str) -> Vec<u8> { (0..s.len()).step_by(2).map(|i| u8::from_str_radix(&s[i..i + 2], 16).unwrap()).collect() }
fn read(p: &str) -> Vec<u8> { if p == "-" { Vec::new() } else { std::fs::read(p).unwrap() } }
fn t(s: SystemTime) -> String { format!("{}", s.duration_since(UNIX_EPOCH).map(|d| d.as_secs() as i64).unwrap_or(-1)) }
fn at(s: &str) -> SystemTime { UNIX_EPOCH + Duration::from_secs(s.parse().unwrap()) }
fn chain(e: &dyn std::error::Error) -> String {
    let mut out = format!("{e}"); let mut s = e.source();
    while let Some(x) = s { out.push_str(&format!(" <- {x}")); s = x.source(); }
    out
}
fn err(e: &Error) -> String { format!("Err({e:?}) display=\"{}\" kind={:?}", chain(e), e.kind()) }

fn main() {
    let a: Vec<String> = std::env::args().collect();
    match a[1].as_str() {
        "req" => {
            let r = if a[4] == "random" { OcspRequest::with_random_nonce(&read(&a[2]), &read(&a[3])) }
                    else { OcspRequest::new(&read(&a[2]), &read(&a[3]), &unhex(&a[4])) };
            match r {
                Ok(r) => { std::fs::write(&a[5], r.to_der().unwrap()).unwrap(); println!("Ok nonce={} ({} octets)", hex(r.nonce()), r.nonce().len()); }
                Err(e) => println!("{}", err(&e)),
            }
        }
        "judge" => {
            // judge <resp> <cert> <issuer> <at> [nonce-hex] [tolerance-secs]
            let resp = match OcspResponse::from_der(&read(&a[2])) { Ok(r) => r, Err(e) => { println!("from_der: {}", err(&e)); return; } };
            println!("from_der: Ok produced_at={} this_update={:?} nonce={:?}", t(resp.produced_at()), resp.this_update().map(t), resp.nonce().map(hex));
            let (c, i, when) = (read(&a[3]), read(&a[4]), at(&a[5]));
            if let Some(n) = a.get(6).filter(|n| *n != "skip") {
                let req = OcspRequest::new(&c, &i, &unhex(n)).unwrap();
                println!("check_nonce({n}): {}", match resp.check_nonce(&req) { Ok(()) => "Ok".into(), Err(e) => err(&e) });
            }
            if let Some(tol) = a.get(7) {
                println!("check_this_update(at={}, tol={tol}s): {}", a[5], match resp.check_this_update(&c, &i, when, Duration::from_secs(tol.parse().unwrap())) { Ok(()) => "Ok".into(), Err(e) => err(&e) });
            }
            println!("status(at={}): {}", a[5], match resp.status(&c, &i, when) {
                Ok(s) => format!("Ok({s:?}){}", match s { nacre_revocation::OcspStatus::Revoked { date, .. } => format!(" date_unix={}", t(date)), _ => String::new() }),
                Err(e) => err(&e) });
        }
        "url" => println!("{:?}", HttpOcspSource::responder_url(&read(&a[2])).map_err(|e| err(&e))),
        "fetch" => {
            // fetch <cert> <issuer> <loopback|public|fallback> [out]
            let (c, i) = (read(&a[2]), read(&a[3]));
            let r = match a[4].as_str() {
                "loopback" => HttpOcspSource::allowing_loopback().revocation(&c, &i),
                "public" => HttpOcspSource::new().revocation(&c, &i),
                "fallback" => Fallback::new(HttpOcspSource::allowing_loopback(), HttpCrlSource::allowing_loopback()).revocation(&c, &i),
                _ => unreachable!(),
            };
            match r {
                Ok(RevocationData::Ocsp(b)) => { println!("Ok(Ocsp, {} bytes)", b.len()); if let Some(o) = a.get(5) { std::fs::write(o, b).unwrap(); } }
                Ok(RevocationData::Crl(b)) => println!("Ok(Crl, {} bytes)", b.len()),
                Ok(_) => println!("Ok(other)"),
                Err(e) => {
                    let inner = std::error::Error::source(&e).and_then(|s| s.downcast_ref::<Error>());
                    println!("Err display=\"{}\" kind={:?} inner={}", chain(&e), e.kind(), inner.map(err).unwrap_or_default());
                }
            }
        }
        _ => unreachable!(),
    }
}

use std::panic::{catch_unwind, AssertUnwindSafe};
use std::time::{Duration, UNIX_EPOCH};
use nacre_revocation::{OcspRequest, OcspResponse};

/// Runs every public entry point on `bytes`; Some(true) when status() says Ok.
fn judge(bytes: &[u8], cert: &[u8], issuer: &[u8], at: u64) -> bool {
    let Ok(r) = OcspResponse::from_der(bytes) else { return false };
    let _ = (r.produced_at(), r.this_update(), r.nonce());
    if let Ok(req) = OcspRequest::new(cert, issuer, &[0x4e; 32]) { let _ = r.check_nonce(&req); }
    let when = UNIX_EPOCH + Duration::from_secs(at);
    let _ = r.check_this_update(cert, issuer, when, Duration::from_secs(300));
    r.status(cert, issuer, when).is_ok()
}

fn main() {
    let a: Vec<String> = std::env::args().collect();
    // hostile <resp> <cert> <issuer> <at> <tbs_start> <tbs_end>
    let orig = std::fs::read(&a[1]).unwrap();
    let (cert, issuer) = (std::fs::read(&a[2]).unwrap(), std::fs::read(&a[3]).unwrap());
    let at: u64 = a[4].parse().unwrap();
    let (ts, te): (usize, usize) = (a[5].parse().unwrap(), a[6].parse().unwrap());
    std::panic::set_hook(Box::new(|_| {}));
    let (mut n, mut panics, mut ok_in_tbs, mut ok_outside) = (0usize, 0usize, 0usize, 0usize);
    let mut run = |m: &[u8], off: Option<usize>| {
        n += 1;
        match catch_unwind(AssertUnwindSafe(|| judge(m, &cert, &issuer, at))) {
            Err(_) => { panics += 1; eprintln!("PANIC at offset {off:?}"); }
            Ok(true) if m != orig.as_slice() => match off { Some(o) if o >= ts && o < te => { ok_in_tbs += 1; eprintln!("Ok inside tbsResponseData at {o}"); } _ => { ok_outside += 1; eprintln!("OKOUT {off:?}"); } },
            _ => {}
        }
    };
    assert!(judge(&orig, &cert, &issuer, at), "the original must judge Ok");
    for len in 0..orig.len() { run(&orig[..len], None); }
    for i in 0..orig.len() {
        for v in [0x00u8, 0x7f, 0x80, 0x81, 0x84, 0xff] { if orig[i] != v { let mut m = orig.clone(); m[i] = v; run(&m, Some(i)); } }
        for bit in 0..8 { let mut m = orig.clone(); m[i] ^= 1 << bit; run(&m, Some(i)); }
    }
    println!("{}: {} bytes, tbsResponseData [{ts},{te}), {n} inputs, panics={panics}, status Ok with a byte changed inside tbsResponseData={ok_in_tbs}, status Ok with a byte changed outside it={ok_outside}", a[1].rsplit('/').next().unwrap(), orig.len());
    // bombs
    let mut bombs: Vec<(String, Vec<u8>)> = vec![("empty".into(), vec![])];
    for depth in [100usize, 10_000, 100_000] {
        let mut d = Vec::new(); for _ in 0..depth { d.extend([0x30, 0x80]); } for _ in 0..depth { d.extend([0, 0]); }
        bombs.push((format!("{depth} nested indefinite SEQUENCEs"), d));
        let mut v: Vec<u8> = vec![];
        for _ in 0..depth.min(10_000) { let l = v.len(); let mut h = vec![0x30, 0x84]; h.extend((l as u32).to_be_bytes()); h.extend(v); v = h; }
        bombs.push((format!("{} nested definite SEQUENCEs", depth.min(10_000)), v));
    }
    bombs.push(("SEQUENCE claiming 2 GiB".into(), vec![0x30, 0x84, 0x80, 0, 0, 0, 0x0a, 0x01, 0x00]));
    bombs.push(("SEQUENCE with 8-octet length".into(), vec![0x30, 0x88, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff, 0xff]));
    bombs.push(("64 MiB of 0x30".into(), vec![0x30; 64 << 20]));
    for (name, b) in bombs {
        let r = catch_unwind(AssertUnwindSafe(|| OcspResponse::from_der(&b).map(|_| ())));
        println!("bomb {name}: {}", match r { Err(_) => "PANIC".to_string(), Ok(Ok(())) => "Ok".into(), Ok(Err(e)) => format!("Err({e})") });
    }
}

# Walk server: s02 bench emulation (CGI script of conformance/pki/cgi-bin/ocsp + /crl/) or the walk PKI responder.
import http.server, subprocess, os, sys
port, kind, base = int(sys.argv[1]), sys.argv[2], sys.argv[3]
class H(http.server.BaseHTTPRequestHandler):
    def log_message(self, fmt, *args):
        with open(os.path.join(base, f"log-{port}.txt"), "a") as f:
            f.write("%s %s\n" % (self.command, self.path) + ("" if not args else "") )
    def send(self, code, ctype, body, extra=()):
        self.send_response(code); self.send_header("Content-Type", ctype)
        for k, v in extra: self.send_header(k, v)
        self.send_header("Content-Length", str(len(body))); self.end_headers(); self.wfile.write(body)
    def do_GET(self):
        p = os.path.join(base, "state", os.path.basename(self.path)) if self.path.startswith("/crl/") else None
        if p and os.path.exists(p): return self.send(200, "application/pkix-crl", open(p, "rb").read())
        self.send(404, "text/plain", b"not found\n")
    def do_POST(self):
        body = self.rfile.read(int(self.headers.get("Content-Length", "0")))
        with open(os.path.join(base, f"last-request-{port}.der"), "wb") as f: f.write(body)
        with open(os.path.join(base, f"log-{port}.txt"), "a") as f: f.write(f"POST {self.path} Content-Type={self.headers.get('Content-Type')} {len(body)} bytes\n")
        if kind == "bench":
            if self.path != "/cgi-bin/ocsp": return self.send(404, "text/plain", b"not found\n")
            env = dict(os.environ, REQUEST_METHOD="POST", CONTENT_LENGTH=str(len(body)), NACRE_PKI_STATE=os.path.join(base, "state"))
            out = subprocess.run(["sh", os.path.join(base, "ocsp.cgi")], input=body, env=env, capture_output=True).stdout
            head, _, payload = out.partition(b"\r\n\r\n")
            hdrs = dict(l.split(": ", 1) for l in head.decode().split("\r\n") if ": " in l)
            code = int(hdrs.get("Status", "200").split()[0])
            return self.send(code, hdrs.get("Content-Type", "application/octet-stream"), payload)
        mode = open(os.path.join(base, "mode")).read().strip()
        if mode == "openssl":
            r = os.path.join(base, "req.der"); o = os.path.join(base, "resp.der"); open(r, "wb").write(body)
            pk = os.path.join(base, "..", "pki")
            subprocess.run(["openssl", "ocsp", "-index", f"{pk}/index.txt", "-CA", f"{pk}/ca.pem", "-rsigner", f"{pk}/ca.pem", "-rkey", f"{pk}/ca.key", "-reqin", r, "-respout", o, "-ndays", "30"], capture_output=True)
            return self.send(200, "application/ocsp-response", open(o, "rb").read())
        if mode.startswith("file:"): return self.send(200, "application/ocsp-response", open(mode[5:], "rb").read())
        if mode == "404": return self.send(404, "text/plain", b"nope\n")
        if mode == "302": return self.send(302, "text/plain", b"", [("Location", "http://127.0.0.1:18393/elsewhere")])
        if mode == "big": return self.send(200, "application/ocsp-response", b"\x30\x83\x01\x00\x00" + b"\x00" * 65536)
        self.send(500, "text/plain", b"bad mode\n")
http.server.ThreadingHTTPServer(("127.0.0.1", port), H).serve_forever()
