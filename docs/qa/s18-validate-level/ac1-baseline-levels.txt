--- src/main.rs
use std::time::SystemTime;
use nacre_validate::{TrustAnchors, ValidationPolicy, validate, validate_with_policy};

fn show(label: &str, pdf: &[u8], anchors: &TrustAnchors) {
    let bench = nacre_revocation::Fallback::new(nacre_revocation::HttpOcspSource::allowing_loopback(), nacre_revocation::HttpCrlSource::allowing_loopback());
    let source: Option<&dyn nacre_core::RevocationSource> = if std::env::var("QA_SOURCE").is_ok() { Some(&bench) } else { None };
    for (name, policy) in [("dss()", ValidationPolicy::dss()), ("strict()", ValidationPolicy::strict())] {
        match validate_with_policy(pdf, anchors, SystemTime::now(), source, &policy) {
            Ok(sigs) => {
                println!("{label} under {name}: {} signature(s)", sigs.len());
                for s in &sigs {
                    println!(
                        "  {}: {:?} / {:?} / level {:?} / best_signature_time {:?}",
                        s.field,
                        s.report.indication(),
                        s.report.sub_indication(),
                        s.report.level(),
                        s.best_signature_time
                    );
                }
            }
            Err(e) => println!("{label} under {name}: Err({e}) kind {:?}", e.kind()),
        }
    }
}

fn main() {
    let args: Vec<String> = std::env::args().collect();
    let root = std::fs::read(&args[1]).unwrap();
    let anchors = TrustAnchors::from_certificates([root.as_slice()]).unwrap();
    for path in &args[2..] {
        let pdf = std::fs::read(path).unwrap();
        show(path, &pdf, &anchors);
    }
    // Invalid inputs a user could plausibly give.
    show("empty input", b"", &anchors);
    show("not a PDF", b"hello, this is not a pdf", &anchors);
    match validate(b"%PDF-1.7\n%%EOF\n") {
        Ok(s) => println!("truncated PDF (validate): Ok, {} signature(s)", s.len()),
        Err(e) => println!("truncated PDF (validate): Err({e}) kind {:?}", e.kind()),
    }
    let pem = std::fs::read_to_string(&args[1].replace("root.der", "root.pem")).unwrap_or_default();
    match TrustAnchors::from_certificates([pem.as_bytes()]) {
        Ok(_) => println!("PEM anchor: accepted"),
        Err(e) => println!("PEM anchor: Err({e}) kind {:?}", e.kind()),
    }
    match TrustAnchors::from_certificates([b"garbage".as_slice()]) {
        Ok(_) => println!("garbage anchor: accepted"),
        Err(e) => println!("garbage anchor: Err({e}) kind {:?}", e.kind()),
    }
}
