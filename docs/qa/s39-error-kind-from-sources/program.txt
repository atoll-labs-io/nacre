QA s39 — the program every transcript runs, quoted whole.

$QA_TMP is the `mktemp -d` directory of the walk (removed at the end); $QA_TMP/copy holds
the files `git ls-files` lists on story/s39-error-kind-from-sources (tree
15c2511ea3c27f115eda5de6b2ffbbdca4a2e94e), where `cargo fetch` (Install) then
`cargo build --workspace --all-features` (Build) ran, both exit 0.

Built with:
  cd $QA_TMP/scratch && NACRE_CRATES=$QA_TMP/copy/crates CARGO_TARGET_DIR=$QA_TMP/target cargo build
Run with:
  cd $QA_TMP/run && $QA_TMP/target/debug/qa-s39 <scenario>

Every server is started by the program itself, in-process, on 127.0.0.1 port 0; a "closed
port" is a port bound then released. No other host is contacted; 10.1.2.3 and [fd00::1]
are refused by the address policy before any connection.

// QA s39: calls the public interface as an application developer would.
use std::error::Error as _;
use std::io::{BufRead, BufReader, Read, Write};
use std::net::TcpListener;
use std::str::FromStr;

use der::oid::ObjectIdentifier;
use der::{DateTime, Encode};
use nacre_core::{DigestAlgorithm, Error, ErrorKind, RevocationSource, TimestampAuthority};
use nacre_revocation::{HttpCrlSource, HttpOcspSource};
use nacre_tsp::HttpTsa;
use p256::ecdsa::{DerSignature, SigningKey};
use sha2::{Digest, Sha256};
use x509_cert::builder::profile::cabf;
use x509_cert::builder::{Builder, CertificateBuilder};
use x509_cert::der::asn1::Ia5String;
use x509_cert::ext::pkix::crl::dp::DistributionPoint;
use x509_cert::ext::pkix::name::{DistributionPointName, GeneralName};
use x509_cert::ext::pkix::{AccessDescription, AuthorityInfoAccessSyntax, CrlDistributionPoints};
use x509_cert::name::Name;
use x509_cert::serial_number::SerialNumber;
use x509_cert::spki::SubjectPublicKeyInfoOwned;
use x509_cert::time::{Time, Validity};

fn corpus(path: &str) -> Vec<u8> {
    std::fs::read(format!("{}/{path}", env!("NACRE_CRATES"))).unwrap()
}

/// Answers every connection on 127.0.0.1 with `status` and `body`, after reading the request.
fn serve(status: &'static str, body: Vec<u8>) -> u16 {
    let listener = TcpListener::bind("127.0.0.1:0").unwrap();
    let port = listener.local_addr().unwrap().port();
    std::thread::spawn(move || {
        for stream in listener.incoming() {
            let mut stream = stream.unwrap();
            let mut reader = BufReader::new(stream.try_clone().unwrap());
            let mut line = String::new();
            let mut length = 0;
            reader.read_line(&mut line).unwrap();
            line.clear();
            while reader.read_line(&mut line).unwrap() > 2 {
                if let Some((n, v)) = line.split_once(':') {
                    if n.eq_ignore_ascii_case("content-length") { length = v.trim().parse().unwrap(); }
                }
                line.clear();
            }
            let mut b = vec![0; length];
            reader.read_exact(&mut b).unwrap();
            let head = format!("HTTP/1.1 {status}\r\nContent-Length: {}\r\nConnection: close\r\n\r\n", body.len());
            stream.write_all(head.as_bytes()).unwrap();
            stream.write_all(&body).unwrap();
        }
    });
    port
}

fn closed_port() -> u16 {
    let l = TcpListener::bind("127.0.0.1:0").unwrap();
    l.local_addr().unwrap().port()
}

fn self_signed(extend: impl FnOnce(&mut CertificateBuilder<cabf::Root>)) -> Vec<u8> {
    let key = SigningKey::from_slice(&[0x31; 32]).unwrap();
    let name = Name::from_str("CN=QA s39,O=Nacre QA,C=FR").unwrap();
    let time = |y| Time::from(DateTime::new(y, 1, 1, 0, 0, 0).unwrap());
    let mut b = CertificateBuilder::new(cabf::Root::new(false, name).unwrap(), SerialNumber::from(1u32),
        Validity::new(time(2026), time(2036)), SubjectPublicKeyInfoOwned::from_key(key.verifying_key()).unwrap()).unwrap();
    extend(&mut b);
    b.build::<_, DerSignature>(&key).unwrap().to_der().unwrap()
}
fn with_crl_point(url: &str) -> Vec<u8> {
    let p = CrlDistributionPoints(vec![DistributionPoint {
        distribution_point: Some(DistributionPointName::FullName(vec![GeneralName::UniformResourceIdentifier(Ia5String::new(url).unwrap())])),
        reasons: None, crl_issuer: None }]);
    self_signed(|b| b.add_extension(&p).unwrap())
}
fn with_responder(url: &str) -> Vec<u8> {
    let a = AuthorityInfoAccessSyntax(vec![AccessDescription {
        access_method: ObjectIdentifier::new_unwrap("1.3.6.1.5.5.7.48.1"),
        access_location: GeneralName::UniformResourceIdentifier(Ia5String::new(url).unwrap()) }]);
    self_signed(|b| b.add_extension(&a).unwrap())
}

fn report<T>(label: &str, result: Result<T, Error>) {
    match result {
        Ok(_) => println!("{label}: Ok(..)"),
        Err(e) => {
            let variant = match &e {
                Error::Timestamp { .. } => "Timestamp { .. }",
                Error::Revocation { .. } => "Revocation { .. }",
                Error::DigestLength { .. } => "DigestLength { .. }",
                Error::Signer(_) => "Signer(..)",
                _ => "other",
            };
            println!("{label}: Err(nacre_core::Error::{variant})");
            println!("  kind()    = {:?} (\"{}\")", e.kind(), e.kind());
            println!("  Display   = {e}");
            let mut s = e.source();
            while let Some(src) = s { println!("  source    = {src}"); s = src.source(); }
            if let Some(src) = e.source() {
                if let Some(t) = src.downcast_ref::<nacre_tsp::Error>() { println!("  downcast  = nacre_tsp::Error, its kind() = {:?}", t.kind()); }
                if let Some(r) = src.downcast_ref::<nacre_revocation::Error>() { println!("  downcast  = nacre_revocation::Error, its kind() = {:?}", r.kind()); }
            }
        }
    }
}

fn main() {
    let scenario = std::env::args().nth(1).unwrap_or_default();
    let data_digest = Sha256::digest(corpus("nacre-tsp/tests/corpus/recorded/data.bin"));
    match scenario.as_str() {
        "core" => {
            report("Error::timestamp(io::Error)", Err::<(), _>(Error::timestamp(std::io::Error::other("tsa down"))));
            report("Error::revocation(\"down\")", Err::<(), _>(Error::revocation("down")));
            report("Error::timestamp_with_kind(InvalidSignature, ..)", Err::<(), _>(Error::timestamp_with_kind(ErrorKind::InvalidSignature, std::io::Error::other("forged"))));
            report("Error::revocation_with_kind(InvalidInput, ..)", Err::<(), _>(Error::revocation_with_kind(ErrorKind::InvalidInput, "bad crl")));
            report("Error::Revocation { kind: Indeterminate, .. } (struct literal)", Err::<(), _>(Error::Revocation { kind: ErrorKind::Indeterminate, source: "x".into() }));
            let e = Error::timestamp(std::io::Error::other("tsa down"));
            println!("source downcasts to io::Error: {}", e.source().unwrap().downcast_ref::<std::io::Error>().is_some());
        }
        "core-error" => {
            // A caller passes a Box<E> instead of E, which the rustdoc warns against.
            let boxed: Box<dyn std::error::Error + Send + Sync> = Box::new(std::io::Error::other("tsa down"));
            let e = Error::timestamp(boxed);
            report("Error::timestamp(Box<io::Error>)", Err::<(), _>(Error::timestamp(Box::new(std::io::Error::other("tsa down")) as Box<dyn std::error::Error + Send + Sync>)));
            println!("source downcasts to io::Error: {}", e.source().unwrap().downcast_ref::<std::io::Error>().is_some());
        }
        "tsa" => {
            let p = serve("200 OK", corpus("nacre-tsp/tests/corpus/attacks/signature-invalid.tsr"));
            report("forged token (signature-invalid.tsr)", HttpTsa::new(&format!("http://127.0.0.1:{p}/tsa")).timestamp(DigestAlgorithm::Sha256, &data_digest));
            let p = serve("200 OK", corpus("nacre-tsp/tests/corpus/attacks/valid.tsr"));
            report("replayed token (valid.tsr, recorded nonce)", HttpTsa::new(&format!("http://127.0.0.1:{p}/tsa")).timestamp(DigestAlgorithm::Sha256, &data_digest));
            let p = serve("200 OK", corpus("nacre-tsp/tests/corpus/attacks/valid.tsr"));
            report("imprint of 31 bytes for SHA-256", HttpTsa::new(&format!("http://127.0.0.1:{p}/tsa")).timestamp(DigestAlgorithm::Sha256, &[0u8; 31]));
            let p = closed_port();
            report("unreachable TSA (closed port)", HttpTsa::new(&format!("http://127.0.0.1:{p}/tsa")).timestamp(DigestAlgorithm::Sha256, &data_digest));
        }
        "tsa-error" => {
            let p = serve("200 OK", b"not a TimeStampResp".to_vec());
            report("TSA answers garbage", HttpTsa::new(&format!("http://127.0.0.1:{p}/tsa")).timestamp(DigestAlgorithm::Sha256, &data_digest));
            let p = serve("500 Internal Server Error", Vec::new());
            report("TSA answers 500", HttpTsa::new(&format!("http://127.0.0.1:{p}/tsa")).timestamp(DigestAlgorithm::Sha256, &data_digest));
            report("URL that is not a URL", HttpTsa::new("not a url").timestamp(DigestAlgorithm::Sha256, &data_digest));
            report("empty imprint", HttpTsa::new("http://127.0.0.1:1/tsa").timestamp(DigestAlgorithm::Sha256, &[]));
        }
        "crl" => {
            let p = serve("200 OK", b"not a CRL".to_vec());
            report("malformed CRL", HttpCrlSource::allowing_loopback().revocation(&with_crl_point(&format!("http://127.0.0.1:{p}/garbage.crl")), &[]));
            let p = serve("200 OK", corpus("nacre-revocation/tests/corpus/openssl/ec-ca.crl"));
            report("refused address (HttpCrlSource::new, 127.0.0.1)", HttpCrlSource::new().revocation(&with_crl_point(&format!("http://127.0.0.1:{p}/ca.crl")), &[]));
            report("refused address (allowing_loopback, 10.1.2.3)", HttpCrlSource::allowing_loopback().revocation(&with_crl_point("http://10.1.2.3/ca.crl"), &[]));
            let p = closed_port();
            report("unreachable server (closed port)", HttpCrlSource::allowing_loopback().revocation(&with_crl_point(&format!("http://127.0.0.1:{p}/ca.crl")), &[]));
            let p = serve("200 OK", corpus("nacre-revocation/tests/corpus/openssl/ec-ca.crl"));
            report("valid CRL, loopback allowed (control)", HttpCrlSource::allowing_loopback().revocation(&with_crl_point(&format!("http://127.0.0.1:{p}/ca.crl")), &[]));
        }
        "crl-error" => {
            report("certificate bytes that are not DER", HttpCrlSource::allowing_loopback().revocation(b"not a certificate", &[]));
            report("certificate without a CRL distribution point", HttpCrlSource::allowing_loopback().revocation(&self_signed(|_| {}), &[]));
            let p = serve("404 Not Found", Vec::new());
            report("server answers 404", HttpCrlSource::allowing_loopback().revocation(&with_crl_point(&format!("http://127.0.0.1:{p}/ca.crl")), &[]));
        }
        "ocsp" => {
            let ca = corpus("nacre-revocation/tests/corpus/ocsp-attacks/ca.der");
            let p = serve("200 OK", corpus("nacre-revocation/tests/corpus/ocsp-attacks/base.der"));
            report("replayed response (fixed nonce)", HttpOcspSource::allowing_loopback().revocation(&with_responder(&format!("http://127.0.0.1:{p}/ocsp")), &ca));
            report("refused address (HttpOcspSource::new, 127.0.0.1)", HttpOcspSource::new().revocation(&with_responder(&format!("http://127.0.0.1:{p}/ocsp")), &ca));
            report("refused address (allowing_loopback, fd00::1)", HttpOcspSource::allowing_loopback().revocation(&with_responder("http://[fd00::1]/ocsp"), &ca));
        }
        "ocsp-error" => {
            let ca = corpus("nacre-revocation/tests/corpus/ocsp-attacks/ca.der");
            report("certificate bytes that are not DER", HttpOcspSource::allowing_loopback().revocation(b"not a certificate", &ca));
            report("certificate without an OCSP responder", HttpOcspSource::allowing_loopback().revocation(&self_signed(|_| {}), &ca));
            let p = serve("200 OK", b"not an OCSP response".to_vec());
            report("responder answers garbage", HttpOcspSource::allowing_loopback().revocation(&with_responder(&format!("http://127.0.0.1:{p}/ocsp")), &ca));
            let p = closed_port();
            report("unreachable responder (closed port)", HttpOcspSource::allowing_loopback().revocation(&with_responder(&format!("http://127.0.0.1:{p}/ocsp")), &ca));
        }
        other => { eprintln!("unknown scenario {other:?}"); std::process::exit(2); }
    }
}
